        if (!(ctx.player() instanceof ServerPlayer player)) return;

        ctx.enqueueWork(() -> {
            ItemStack stack;
            if (player.containerMenu instanceof GlyphWorkbenchMenu menu) {
                stack = menu.getGrimoire();
            } else {
                int slot = pkt.itemSlot();
                if (slot < 0 || slot >= player.getInventory().getContainerSize()) return;
                stack = player.getInventory().getItem(slot);
            }

            if (!(stack.getItem() instanceof GrimoireItem)) return;

            int[] a = pkt.ints();
            String[] s = pkt.strings();

            switch (pkt.action()) {
                case "setPage"           -> { if (a.length < 1) return; }
                case "switchPage"        -> { if (a.length < 1) return; }
                case "placeGlyph"        -> { if (a.length < 2 || s.length < 1) return; }
                case "moveGlyph"         -> { if (a.length < 3) return; }
                case "removeGlyph"       -> { if (a.length < 2) return; }
                case "setName"           -> { if (a.length < 1 || s.length < 1) return; }
                case "setFavorite"       -> { if (a.length < 2) return; }
                case "movePage"          -> { if (a.length < 2) return; }
                case "resetGlyph"        -> { if (a.length < 2) return; }
                case "resetGlyphInBank"  -> { if (a.length < 1) return; }
                default -> { }
            }

            switch (pkt.action()) {
                case "setPage"     -> GrimoireData.setActivePage(stack, a[0]);
                case "switchPage"  -> { GrimoireData.switchPage(stack, a[0] != 0); broadcastTitle(player, stack); }
                case "placeGlyph"  -> {
                    Map<Stat, Integer> allocs = new HashMap<>();
                    if (s.length > 1 && !s[1].isEmpty()) {
                        for (String pair : s[1].split(",")) {
                            String[] kv = pair.split(":");
                            if (kv.length != 2) continue;
                            try {
                                int v = Integer.parseInt(kv[1]);
                                allocs.put(Stat.valueOf(kv[0]), v);
                            } catch (Exception ignored) {}
                        }
                    }
                    GrimoireData.placeGlyphWithAllocs(stack, a[0], a[1], s[0], allocs);
                }
                case "moveGlyph"   -> GrimoireData.moveGlyph(stack, a[0], a[1], a[2]);
                case "removeGlyph" -> GrimoireData.removeGlyph(stack, a[0], a[1]);
                case "setName"     -> GrimoireData.setPageName(stack, a[0], s[0]);
                case "setFavorite" -> GrimoireData.setPageFavorite(stack, a[0], a[1] != 0);
                case "movePage"    -> { GrimoireData.movePage(stack, a[0], a[1]); broadcastTitle(player, stack); }
                case "resetGlyph"  -> GrimoireData.resetGlyph(stack, a[0], a[1]);
                case "resetGlyphInBank" -> GrimoireData.resetGlyphInBank(stack, a[0]);
                default -> { }
            }
            player.containerMenu.broadcastChanges();
        });
    }

    private static void broadcastTitle(ServerPlayer player, ItemStack stack) {
        int idx = GrimoireData.getActivePage(stack);
        GlyphPage page = GrimoireData.getPage(stack, idx);
        Component msg;
        if (page == null) msg = Component.translatable("gui.grimoiremod.action.page", idx + 1);
        else if (page.isToc()) msg = Component.translatable("gui.grimoiremod.action.toc");
        else {
            int n = GrimoireData.spellNumber(stack, idx);
            String name = page.name == null ? "" : page.name.trim();
            msg = name.isEmpty()
                ? Component.translatable("gui.grimoiremod.action.spell", n)
                : Component.translatable("gui.grimoiremod.action.spell_named", n, name);
        }
        player.displayClientMessage(msg, true);
    }
}

package com.example.grimoiremod.network;

import com.example.grimoiremod.data.SkillTreeData;
import com.example.grimoiremod.data.SkillTreeDef;
import com.example.grimoiremod.item.GrimoireItem;
import com.example.grimoiremod.util.GrimoireLocator;
import net.minecraft.network.FriendlyByteBuf;
import net.minecraft.network.codec.StreamCodec;
import net.minecraft.network.protocol.common.custom.CustomPacketPayload;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.item.ItemStack;
import net.neoforged.neoforge.network.handling.IPayloadContext;

public record LevelUpSkillNodePacket(int itemSlot, String nodeId) implements CustomPacketPayload {

    public static final Type<LevelUpSkillNodePacket> TYPE =
            new Type<>(ResourceLocation.fromNamespaceAndPath("grimoiremod", "level_up_skill_node"));

    public static final StreamCodec<FriendlyByteBuf, LevelUpSkillNodePacket> CODEC = StreamCodec.of(
        (buf, pkt) -> {
            buf.writeVarInt(pkt.itemSlot());
            buf.writeUtf(pkt.nodeId());
        },
        buf -> new LevelUpSkillNodePacket(buf.readVarInt(), buf.readUtf(64))
    );

    @Override public Type<? extends CustomPacketPayload> type() { return TYPE; }

    public static void handle(LevelUpSkillNodePacket pkt, IPayloadContext ctx) {
        if (!(ctx.player() instanceof ServerPlayer player)) return;
        ctx.enqueueWork(() -> {
            int actualSlot = pkt.itemSlot();
            if (actualSlot < 0) actualSlot = GrimoireLocator.findFirst(player);
            if (actualSlot < 0 || actualSlot >= player.getInventory().getContainerSize()) return;

            ItemStack stack = player.getInventory().getItem(actualSlot);
            if (!(stack.getItem() instanceof GrimoireItem)) return;

            SkillTreeDef.Node node = SkillTreeDef.byId(pkt.nodeId());
            if (node == null || !SkillTreeData.isUnlocked(stack, node.id())) return;

            int level = SkillTreeData.getNodeLevel(stack, node.id());
            if (level >= node.maxLevel()) return;
            if (SkillTreeData.getPointsFrac(stack) < 1.0f) return;

            SkillTreeData.addPoints(stack, -1.0f);
            SkillTreeData.addNodeLevel(stack, node.id(), 1);
            player.containerMenu.broadcastChanges();
        });
    }
}

package com.example.grimoiremod.network;

import com.example.grimoiremod.data.SkillTreeData;
import com.example.grimoiremod.data.SkillTreeDef;
import com.example.grimoiremod.item.GrimoireItem;
import net.minecraft.ChatFormatting;
import net.minecraft.network.FriendlyByteBuf;
import net.minecraft.network.chat.Component;
import net.minecraft.network.codec.StreamCodec;
import net.minecraft.network.protocol.common.custom.CustomPacketPayload;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.item.ItemStack;
import net.neoforged.neoforge.network.handling.IPayloadContext;

public record UnlockSkillNodePacket(int itemSlot, String nodeId) implements CustomPacketPayload {

    public static final Type<UnlockSkillNodePacket> TYPE =
            new Type<>(ResourceLocation.fromNamespaceAndPath("grimoiremod", "unlock_skill_node"));

    public static final StreamCodec<FriendlyByteBuf, UnlockSkillNodePacket> CODEC = StreamCodec.of(
        (buf, pkt) -> {
            buf.writeVarInt(pkt.itemSlot());
            buf.writeUtf(pkt.nodeId());
        },
        buf -> new UnlockSkillNodePacket(buf.readVarInt(), buf.readUtf(64))
    );

    @Override public Type<? extends CustomPacketPayload> type() { return TYPE; }

    public static UnlockSkillNodePacket of(int itemSlot, String nodeId) {
        return new UnlockSkillNodePacket(itemSlot, nodeId);
    }

    public static void handle(UnlockSkillNodePacket pkt, IPayloadContext ctx) {
        if (!(ctx.player() instanceof ServerPlayer player)) return;
        ctx.enqueueWork(() -> {
            if (pkt.itemSlot() < 0 || pkt.itemSlot() >= player.getInventory().getContainerSize()) return;
            ItemStack stack = player.getInventory().getItem(pkt.itemSlot());
            if (!(stack.getItem() instanceof GrimoireItem)) return;
            SkillTreeDef.Node node = SkillTreeDef.byId(pkt.nodeId());
            if (node == null) return;

            if (SkillTreeData.isUnlocked(stack, node.id())) return;
            if (SkillTreeData.getPoints(stack) < node.cost()) {
                player.sendSystemMessage(
                        Component.translatable("gui.grimoiremod.skill_tree.no_points")
                                .withStyle(ChatFormatting.RED));
                return;
            }
            boolean ok = SkillTreeData.tryUnlock(stack, node.id(), node.cost(), node.prereqs());
            if (ok) {
                player.sendSystemMessage(
                        Component.translatable("gui.grimoiremod.skill_tree.unlocked")
                                .withStyle(ChatFormatting.GREEN));
            }
            player.containerMenu.broadcastChanges();
        });
    }
}

package com.example.grimoiremod.util;

import com.example.grimoiremod.item.GrimoireItem;
import net.minecraft.world.entity.player.Player;

public final class GrimoireLocator {
    private GrimoireLocator() {}

    /** Возвращает слот активного гримуара: selected если гримуар, иначе первый в инвентаре. -1 = нет. */
    public static int findFirst(Player player) {
        var inv = player.getInventory();
        if (inv.getItem(inv.selected).getItem() instanceof GrimoireItem) return inv.selected;
        for (int i = 0; i < inv.getContainerSize(); i++) {
            if (inv.getItem(i).getItem() instanceof GrimoireItem) return i;
        }
        return -1;
    }
}
