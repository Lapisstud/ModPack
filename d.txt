plugins {

base { archivesName = 'grimoiremod' }

java.toolchain.languageVersion = JavaLanguageVersion.of(21)

    client {
        systemProperty 'neoforge.enabledGameTestNamespaces', 'grimoiremod'

    }
    server {
        systemProperty 'neoforge.enabledGameTestNamespaces', 'grimoiremod'
        systemProperties.put('ignoreList',
            'bootstraplauncher,securejarhandler,asm-commons,asm-util,asm-analysis,asm-tree,asm,JarJarFileSystems,client-extra,fmlcore,javafmllanguage')
    }
}

repositories {
    mavenLocal()
    mavenCentral()
    maven { url = 'https://neoforged.forgecdn.net/releases' } // Зеркало NeoForged
    maven { url = 'https://repo.spongepowered.org/repository/forge-proxy/' } // Для groovydslimprover
    maven { url = 'https://maven.neoforged.net/releases' } // Официальный (на случай, если заработает)
}

dependencies {
    implementation "net.neoforged:neoforge:21.1.218"

        }
    }
}
