        Map<Integer, String> errors = parsed != null ? parsed.errors() : Map.of();

        int slotsToRender = Math.min(page.slotCount, MAX_SLOTS_PER_PAGE);
        for (int i = 0; i < slotsToRender; i++) {
            int x = startX;
            int y = startY + i * (SLOT + SLOT_GAP);
            boolean selected = i == selectedSlot;
            boolean hover = inRect(mx, my, x, y, SLOT, SLOT);

            int border = selected ? SLOT_SEL : SLOT_BORDER;
            gfx.fill(x, y, x + SLOT, y + SLOT, border);
            int bg = hover ? 0xFFFFF4C0 : 0xFFE8DDC0;
            gfx.fill(x + 1, y + 1, x + SLOT - 1, y + SLOT - 1, bg);

            GlyphInstance inst = page.slots[i];
            if (inst != null) {
                Glyph g = Glyphs.byId(inst.typeId());
                if (g != null) {
                    ClientGlyphIcons.draw(gfx, font, g, x, y, SLOT);
                    if (errors.containsKey(i)) {
                        gfx.fill(x - 1, y - 1, x + 9, y + 9, 0xFFFF0000);
                        gfx.drawString(font, "!", x + 4 - font.width("!") / 2, y, 0xFFFFFFFF, false);
                        if (hover) hoverError = errors.get(i);
                    }
                    if (hover) hoverGlyph = inst;
                }
            }
        }

        if (errors.containsKey(-1) && inRect(mx, my, startX, startY, SLOT, slotsToRender * (SLOT + SLOT_GAP))) {
            if (hoverGlyph == null && hoverError == null) hoverError = errors.get(-1);
        }

        int resX = leftPageX + 55;
        int resY = leftPageY + 55;
        gfx.drawString(font, Component.translatable("gui.grimoiremod.result").getString(), resX, resY, INK_DIM, false);

        List<Component> resultLines = snap.describedLines();
        if (resultLines == null) resultLines = List.of();
        int lineY = resY + 12;
        for (Component line : resultLines) {
            List<FormattedCharSequence> wrapped = font.split(line, PAGE_W - 65);
            for (FormattedCharSequence seq : wrapped) {
                if (lineY + 10 > leftPageY + PAGE_H - 20) break;
                gfx.drawString(font, seq, resX, lineY, INK, false);
                lineY += 10;
            }
        }
    }

    private void renderBank(GuiGraphics gfx, int mx, int my, Snapshot snap) {
        gfx.drawString(font, Component.translatable("gui.grimoiremod.bank").getString(),
                rightPageX + 8, rightPageY + 4, CAT_HEADER, false);

        int startX = rightPageX + 8;
        int startY = rightPageY + 38;

        Map<GlyphInstance, Integer> bank = snap.bank();
        if (bank == null) return;

        List<Map.Entry<GlyphInstance, Integer>> filtered = new ArrayList<>();
        for (var e : bank.entrySet()) {
            Glyph g = Glyphs.byId(e.getKey().typeId());
            if (g == null) continue;
            if (selectedBankTab != null && g.category() != selectedBankTab) continue;
            filtered.add(e);
        }

        int bankAreaH = PAGE_H - 38 - 6;
        int maxVisible = BANK_COLS * (bankAreaH / (BANK_CELL + BANK_GAP));
        int maxScroll = Math.max(0, filtered.size() - maxVisible);
        if (bankScroll > maxScroll) bankScroll = maxScroll;
        if (bankScroll < 0) bankScroll = 0;

        for (int gi = bankScroll; gi < filtered.size() && gi < bankScroll + maxVisible; gi++) {
            var e = filtered.get(gi);
            GlyphInstance inst = e.getKey();
            Glyph g = Glyphs.byId(inst.typeId());
            if (g == null) continue;

            int localIdx = gi - bankScroll;
            int col = localIdx % BANK_COLS;
            int row = localIdx / BANK_COLS;

            int y = startY + row * (BANK_CELL + BANK_GAP);

            boolean hover = inRect(mx, my, x, y, ICON + 4, ICON + 4);
            if (hover && draggingInst == null) {
                hoverGlyph = inst;
                hoverError = null;
            }

            int border = hover ? 0xFFFFFFFF : SLOT_BORDER;
            gfx.fill(x - 1, y - 1, x + ICON + 3, y + ICON + 3, border);
            gfx.fill(x, y, x + ICON + 2, y + ICON + 2, 0xFFF8EFD5);
            ClientGlyphIcons.draw(gfx, font, g, x + 1, y + 1, ICON);

            String cnt = "x" + e.getValue();
            int tw = font.width(cnt);
            int cx = x + ICON + 2 - tw + 1;
            int cy = y + ICON + 2 - 8;
            gfx.drawString(font, cnt, cx, cy, 0xFF333333, false);

            if (inst.totalPoints() > 0) {
                gfx.drawString(font, "+" + inst.totalPoints(), x + 1, y - 6, 0xFFFFAA00, false);
            }
        }

        if (filtered.size() > maxVisible) {
            int barX = rightPageX + PAGE_W - 5;
            int barY = startY;
            gfx.fill(barX, barY, barX + 2, barY + bankAreaH, 0x33FFFFFF);
            int handleH = Math.max(10, bankAreaH * maxVisible / filtered.size());
            int handleY = barY + (bankAreaH - handleH) * bankScroll / Math.max(1, maxScroll);
            gfx.fill(barX, handleY, barX + 2, handleY + handleH, 0xFFCCCCCC);
        }

        if (filtered.isEmpty()) {
            String msg = Component.translatable("gui.grimoiremod.no_glyphs").getString();
            int tw = font.width(msg);
            gfx.drawString(font, msg, rightPageX + PAGE_W / 2 - tw / 2, startY + 40, INK_DIM, false);
        }

    private void renderNav(GuiGraphics gfx, int mx, int my, int active, int total, Snapshot snap) {

    private void renderTabs(GuiGraphics gfx, int mx, int my) {
        int startX = rightPageX + 4;
        int startY = rightPageY + 20;
        int tabH = 14;
        int activeIdx = getTabIndex(selectedBankTab);
        int expandedIdx = (hoverBankTab != -1) ? hoverBankTab : activeIdx;
        int currentX = startX;

        for (int t = 0; t < TAB_CATEGORIES.length; t++) {
            String label = (t == expandedIdx)
                    ? Component.translatable(TAB_LABELS_FULL[t]).getString()
                    : Component.translatable(TAB_LABELS_SHORT[t]).getString();
            int w = font.width(label) + 6;

            boolean isActive = (t == activeIdx);
            boolean isHover = (t == hoverBankTab);
            int color = isActive ? TAB_ACTIVE : (isHover ? TAB_HOVER : TAB_IDLE);

            gfx.fill(currentX, startY, currentX + w, startY + tabH, color);
            gfx.fill(currentX, startY + tabH, currentX + w, startY + tabH + 1, SLOT_BORDER);
            if (isActive) gfx.fill(currentX, startY + tabH - 2, currentX + w, startY + tabH, TOC_GOLD);

            gfx.drawString(font, label, currentX + 3, startY + 3, INK, false);

            tabPositions[t] = new int[]{currentX, startY, w, tabH};
            currentX += w + 1;
        }

    private int slotUnderCursor(double mx, double my, GlyphPage page) {

            int y = startY + i * (SLOT + SLOT_GAP);
            if (inRect(mx, my, startX, y, SLOT, SLOT)) return i;

    private int getTabIndex(GlyphCategory cat) {
        for (int i = 0; i < TAB_CATEGORIES.length; i++) {
            if (TAB_CATEGORIES[i] == cat) return i;
        }
        return 0;
    }

    private void refreshHoverState(double mx, double my) {
        hoverBookmark = -1;
        hoverNav = 0;
        Snapshot snap = snapshot();
        if (snap != null) {
            int bx = bookLeft + BOOK_W + COVER_PAD + 2;
            int by0 = bookTop + 8;
            int bh = (BOOK_H - 16) / 10;
            int favCount = 0;
            for (int i = 0; i < snap.pageCount() && favCount < 10; i++) {
                if (snap.page(i) == null || !snap.page(i).favorite) continue;
                if (inRect(mx, my, bx, by0 + favCount * bh, 22, bh - 1)) { hoverBookmark = i; break; }
                favCount++;
            }
            int y = bookTop + BOOK_H - 14;
            if (inRect(mx, my, leftPageX + 6, y, 14, 12)) hoverNav = -1;
            if (inRect(mx, my, rightPageX + PAGE_W - 20, y, 14, 12)) hoverNav = +1;
        }
        updateTabHover(mx, my);
    }

    private void updateTabHover(double mx, double my) {
        int newHover = -1;
        for (int t = 0; t < TAB_CATEGORIES.length; t++) {
            if (tabPositions[t][2] > 0 && inRect(mx, my, tabPositions[t][0], tabPositions[t][1], tabPositions[t][2], tabPositions[t][3])) {
                newHover = t;
                break;
            }
        }
        hoverBankTab = newHover;
    }

    @Override
    public boolean mouseClicked(double mx, double my, int button) {
        Snapshot snap = snapshot();
        boolean onToc = snap == null || snap.activePage() == null || snap.activePage().isToc();
        if (!onToc && nameField != null && nameField.isMouseOver(mx, my)) {
            nameField.setFocused(true);
            nameField.mouseClicked(mx, my, button);
            return true;
        }
        if (snap == null) return super.mouseClicked(mx, my, button);

        boolean wasFocused = nameField != null && nameField.isFocused();
        if (nameField != null) nameField.setFocused(false);
        if (wasFocused && snap.activePage() != null && snap.activePage().isSpell()) {
            sendName(snap.active(), snap.activePage());
        }

        refreshHoverState(mx, my);

        int active = snap.active();
        GlyphPage page = snap.activePage();
        if (page == null) return super.mouseClicked(mx, my, button);

        if (hoverBookmark >= 0) {
            if (hoverBookmark != active) {
                sendName(active, page);
                PacketDistributor.sendToServer(GrimoireActionPacket.setPage(itemSlotId, hoverBookmark));
                selectedSlot = -1;
                draggingInst = null;
            }
            return true;
        }

        if (hoverNav != 0) {
            sendName(active, page);
            PacketDistributor.sendToServer(GrimoireActionPacket.switchPage(itemSlotId, hoverNav > 0));
            selectedSlot = -1;
            draggingInst = null;
            return true;
        }

        if (!page.isToc()) {
            if (inRect(mx, my, leftPageX + PAGE_W - 20, leftPageY + 4, 12, 12)) {
                PacketDistributor.sendToServer(GrimoireActionPacket.setFavorite(itemSlotId, active, !page.favorite));
                return true;
            }
            for (int t = 0; t < TAB_CATEGORIES.length; t++) {
                if (tabPositions[t][2] > 0 && inRect(mx, my, tabPositions[t][0], tabPositions[t][1], tabPositions[t][2], tabPositions[t][3])) {
                    selectedBankTab = TAB_CATEGORIES[t];
                    return true;
                }
            }
        }

        if (page.isToc()) {
            int count = snap.pageCount();
            int rowY = leftPageY + TOC_TOP;
            for (int i = 0; i < count; i++) {
                GlyphPage p = snap.page(i);
                if (p == null || p.isToc()) continue;
                if (inRect(mx, my, leftPageX + PAGE_W - 20, rowY, 12, TOC_ROW_H - 2)) {
                    PacketDistributor.sendToServer(GrimoireActionPacket.setFavorite(itemSlotId, i, !p.favorite));
                    return true;
                }
                if (inRect(mx, my, leftPageX + 8, rowY, PAGE_W - 16, TOC_ROW_H - 2)) {
                    resetTocDrag();
                    tocDragFrom = i;
                    selectedSlot = -1;
                    draggingInst = null;
                    return true;
                }
                rowY += TOC_ROW_H;
            }
            return super.mouseClicked(mx, my, button);
        }

        Map<GlyphInstance, Integer> bank = snap.bank();
        if (bank != null) {
            int startX = rightPageX + 8;
            int startY = rightPageY + 38;
            List<Map.Entry<GlyphInstance, Integer>> filtered = new ArrayList<>();
            for (var e : bank.entrySet()) {
                Glyph g = Glyphs.byId(e.getKey().typeId());
                if (g == null) continue;
                if (selectedBankTab != null && g.category() != selectedBankTab) continue;
                filtered.add(e);
            }
            int bankAreaH = PAGE_H - 38 - 6;
            int maxVisible = BANK_COLS * (bankAreaH / (BANK_CELL + BANK_GAP));
            for (int gi = bankScroll; gi < filtered.size() && gi < bankScroll + maxVisible; gi++) {
                int localIdx = gi - bankScroll;
                int col = localIdx % BANK_COLS;
                int row = localIdx / BANK_COLS;
                int x = startX + col * (BANK_CELL + BANK_GAP);
                int y = startY + row * (BANK_CELL + BANK_GAP);
                if (inRect(mx, my, x, y, ICON + 4, ICON + 4)) {
                    draggingInst = filtered.get(gi).getKey();
                    draggingFromPage = false;
                    return true;
                }
            }
        }

        int startY = leftPageY + 70;
        int startX = leftPageX + 15;
        for (int i = 0; i < Math.min(page.slotCount, MAX_SLOTS_PER_PAGE); i++) {
            int y = startY + i * (SLOT + SLOT_GAP);
            if (inRect(mx, my, startX, y, SLOT, SLOT)) {
                if (button == 1) {
                    if (page.slots[i] != null) {
                        PacketDistributor.sendToServer(GrimoireActionPacket.removeGlyph(itemSlotId, active, i));
                    }
                } else {
                    if (page.slots[i] != null) {
                        selectedSlot = (selectedSlot == i) ? -1 : i;
                        draggingInst = page.slots[i];
                        draggingFromPage = true;
                    } else {
                        selectedSlot = -1;
                    }
                }
                return true;
            }
        }

        return super.mouseClicked(mx, my, button);
    }

    @Override
    public boolean mouseScrolled(double mx, double my, double dx, double dy) {
        Snapshot snap = snapshot();
        if (snap == null) return super.mouseScrolled(mx, my, dx, dy);
        GlyphPage activePage = snap.activePage();
        if (activePage == null || activePage.isToc() || snap.bank() == null) {
            return super.mouseScrolled(mx, my, dx, dy);
        }
        if (!inRect(mx, my, rightPageX, rightPageY + 38, PAGE_W, PAGE_H - 38)) {
            return super.mouseScrolled(mx, my, dx, dy);
        }
        int bankAreaH = PAGE_H - 38 - 6;
        int maxVisible = BANK_COLS * (bankAreaH / (BANK_CELL + BANK_GAP));
        int visible = 0;
        for (var e : snap.bank().entrySet()) {
            Glyph g = Glyphs.byId(e.getKey().typeId());
            if (g == null) continue;
            if (selectedBankTab != null && g.category() != selectedBankTab) continue;
            visible++;
        }
        int maxScroll = Math.max(0, visible - maxVisible);
        bankScroll = (int)Math.max(0, Math.min(maxScroll, bankScroll - Math.signum(dy)));
        return true;
    }

    @Override
    public boolean mouseDragged(double mx, double my, int button, double dx, double dy) {
        if (tocDragFrom >= 0) {
            Snapshot snap = snapshot();
            if (snap != null) updateTocDrop(mx, my, snap);
            return true;
        }
        return super.mouseDragged(mx, my, button, dx, dy);
    }

    @Override
    public boolean mouseReleased(double mx, double my, int button) {
        if (tocDragFrom >= 0) {
            int from = tocDragFrom;
            int target = tocDropTarget;
            boolean after = tocDropAfter;
            boolean moved = tocDragMoved;
            resetTocDrag();
            if (!moved) {
                PacketDistributor.sendToServer(GrimoireActionPacket.setPage(itemSlotId, from));
                selectedSlot = -1;
            } else if (target >= 0) {
                int to = computeToPage(from, target, after);
                if (to != from) {
                    PacketDistributor.sendToServer(GrimoireActionPacket.movePage(itemSlotId, from, to));
                }
            }
            return true;
        }

        if (draggingInst != null) {
            GlyphInstance drag = draggingInst;
            draggingInst = null;
            Snapshot snap = snapshot();
            if (snap != null) {
                int active = snap.active();
                GlyphPage page = snap.activePage();
                if (page != null && !page.isToc()) {
                    int dropSlot = slotUnderCursor(mx, my, page);
                    if (draggingFromPage) {
                        if (dropSlot < 0) {
                            PacketDistributor.sendToServer(GrimoireActionPacket.removeGlyph(itemSlotId, active, selectedSlot));
                            selectedSlot = -1;
                        } else if (dropSlot != selectedSlot && page.slots[dropSlot] == null) {
                            PacketDistributor.sendToServer(GrimoireActionPacket.moveGlyph(itemSlotId, active, selectedSlot, dropSlot));
                            selectedSlot = -1;
                        }
                    } else if (dropSlot >= 0 && page.slots[dropSlot] == null) {
                        PacketDistributor.sendToServer(
                                GrimoireActionPacket.placeGlyph(itemSlotId, active, dropSlot, drag.typeId(), drag.allocations()));
                    } else if (selectedSlot >= 0 && selectedSlot < page.slotCount && page.slots[selectedSlot] == null) {
                        PacketDistributor.sendToServer(
                                GrimoireActionPacket.placeGlyph(itemSlotId, active, selectedSlot, drag.typeId(), drag.allocations()));
                        selectedSlot = -1;
                    }
                }
            }
            draggingFromPage = false;
            return true;
        }
        return super.mouseReleased(mx, my, button);
    }

    @Override
    public boolean keyPressed(int keyCode, int scanCode, int modifiers) {
        if (nameField != null && nameField.isFocused()) {
            if (keyCode == GLFW.GLFW_KEY_ENTER || keyCode == GLFW.GLFW_KEY_KP_ENTER) {
                Snapshot snap = snapshot();
                if (snap != null) sendName(snap.active(), snap.activePage());
                nameField.setFocused(false);
                return true;
            }
            if (nameField.keyPressed(keyCode, scanCode, modifiers)) return true;
        }
        if (keyCode == GLFW.GLFW_KEY_ESCAPE) {
            if (tocDragFrom >= 0) { resetTocDrag(); return true; }
            if (draggingInst != null) { draggingInst = null; return true; }
            onClose();
            return true;
        }
        return super.keyPressed(keyCode, scanCode, modifiers);
    }

    @Override
    public boolean charTyped(char ch, int modifiers) {
        if (nameField != null && nameField.isFocused() && nameField.charTyped(ch, modifiers)) return true;
        return super.charTyped(ch, modifiers);
    }

    @Override
    public void onClose() {
        resetTocDrag();
        draggingInst = null;
        Snapshot snap = snapshot();
        if (snap != null) sendName(snap.active(), snap.activePage());
        super.onClose();
    }

    private void sendName(int active, GlyphPage page) {
        if (nameField == null || page == null || !page.isSpell()) return;
        String name = nameField.getValue();
        String old = page.name == null ? "" : page.name;
        if (!name.equals(old)) {
            PacketDistributor.sendToServer(GrimoireActionPacket.setName(itemSlotId, active, name));
        }
    }

    private static boolean inRect(double mx, double my, int x, int y, int w, int h) {
        return mx >= x && mx < x + w && my >= y && my < y + h;
    }

    private static int brighten(int color) {
        int a = (color >> 24) & 0xFF;
        int r = Math.min(255, ((color >> 16) & 0xFF) + 30);
        int g = Math.min(255, ((color >>  8) & 0xFF) + 30);
        int b = Math.min(255, ( color        & 0xFF) + 30);
        return (a << 24) | (r << 16) | (g << 8) | b;
    }

    private void updateTocDrop(double mx, double my, Snapshot snap) {
        int count = snap.pageCount();
        int rowY = leftPageY + TOC_TOP;
        int found = -1;
        boolean after = false;

        for (int i = 0; i < count; i++) {
            GlyphPage p = snap.page(i);
            if (p == null || p.isToc()) continue;
            int top = rowY;
            int bottom = rowY + TOC_ROW_H - 2;
            if (my >= top && my < bottom) {
                found = i;
                after = my >= (top + bottom) / 2.0;
                break;
            }
            rowY += TOC_ROW_H;
        }

        if (found < 0) {
            if (my < leftPageY + TOC_TOP) {
                for (int i = 0; i < count; i++) {
                    GlyphPage p = snap.page(i);
                    if (p != null && !p.isToc()) { found = i; after = false; break; }
                }
            } else {
                for (int i = count - 1; i >= 0; i--) {
                    GlyphPage p = snap.page(i);
                    if (p != null && !p.isToc()) { found = i; after = true; break; }
                }
            }
        }

        tocDropTarget = found;
        tocDropAfter = after;
        if (found != -1 && found != tocDragFrom) tocDragMoved = true;
    }

    private int computeToPage(int from, int tgt, boolean after) {
        if (tgt == from) return from;
        if (after) return (tgt > from) ? tgt : tgt + 1;
        else       return (tgt > from) ? tgt - 1 : tgt;
    }

    private void resetTocDrag() {
        tocDragFrom = -1;
        tocDropTarget = -1;
        tocDropAfter = false;
        tocDragMoved = false;
    }
}

    private ItemStack book() {
        var p = Minecraft.getInstance().player;
        if (p == null) return ItemStack.EMPTY;
        int actualSlot = itemSlotId >= 0 ? itemSlotId : GrimoireLocator.findFirst(p);
        if (actualSlot < 0 || actualSlot >= p.getInventory().getContainerSize()) return ItemStack.EMPTY;
        ItemStack s = p.getInventory().getItem(actualSlot);
        return s.getItem() instanceof GrimoireItem ? s : ItemStack.EMPTY;
    }

    @Override public boolean isPauseScreen() { return false; }

    @Override
    public void render(GuiGraphics gfx, int mx, int my, float pt) {
        super.render(gfx, mx, my, pt);
        gfx.fill(0, 0, width, height, 0x80000000); // Полупрозрачный фон

        renderTabs(gfx, mx, my);

        if (activeTab == TAB_GENERAL) {
            String soon = Component.translatable("gui.grimoiremod.skill_tree.soon").getString();
            gfx.drawString(font, soon, width / 2 - font.width(soon) / 2, height / 2, 0xFFAAAAAA, false);
            return;
        }

        ItemStack book = book();
        boolean hasBook = !book.isEmpty();
        float ptsFrac = hasBook ? SkillTreeData.getPointsFrac(book) : 0f;
        
        String ptsStr = String.valueOf((int)ptsFrac);
        if (ptsFrac - (int)ptsFrac > 0.01f) ptsStr = String.format("%.1f", ptsFrac);

        gfx.pose().pushPose();
        gfx.pose().translate(width / 2.0f + panX, height / 2.0f + panY, 0);
        gfx.pose().scale(zoom, zoom, 1.0f);

        hoverNodeId = null;

            for (String preId : n.prereqs()) {
                SkillTreeDef.Node pre = SkillTreeDef.byId(preId);
                if (pre == null) continue;
                
                boolean unlocked = hasBook && SkillTreeData.isUnlocked(book, n.id());
                int color = unlocked ? LINE_UNLOCKED : LINE_LOCKED;
                
                drawLine(gfx, pre.x() * SPACING, pre.y() * SPACING, n.x() * SPACING, n.y() * SPACING, color);
            }
        }

        // 2. Узлы
        for (SkillTreeDef.Node n : SkillTreeDef.NODES) {
            int nx = n.x() * SPACING - NODE_SIZE / 2;
            int ny = n.y() * SPACING - NODE_SIZE / 2;
            
            boolean unlocked = hasBook && SkillTreeData.isUnlocked(book, n.id());
            boolean canUnlock = false;
            if (hasBook && !unlocked) {
                boolean hasPrereqs = true;
                for (String p : n.prereqs()) {
                    if (!SkillTreeData.isUnlocked(book, p)) { hasPrereqs = false; break; }
                }
                if (hasPrereqs && ptsFrac >= n.cost()) canUnlock = true;
            }

            double worldMx = (mx - (width / 2.0f + panX)) / zoom;
            double worldMy = (my - (height / 2.0f + panY)) / zoom;
            boolean hover = worldMx >= nx && worldMx < nx + NODE_SIZE && worldMy >= ny && worldMy < ny + NODE_SIZE;
            if (hover) hoverNodeId = n.id();

            int border = unlocked ? 0xFF55AA55 : (canUnlock ? 0xFFFFFF55 : 0xFF555555);
            int fill = unlocked ? 0xFF2A5A2A : (canUnlock ? 0xFF5A5A2A : 0xFF2A2A2A);
            
            if (hover) border = 0xFFFFFFFF;

            gfx.fill(nx - 2, ny - 2, nx + NODE_SIZE + 2, ny + NODE_SIZE + 2, 0xFF000000);
            gfx.fill(nx, ny, nx + NODE_SIZE, ny + NODE_SIZE, border);
            gfx.fill(nx + 2, ny + 2, nx + NODE_SIZE - 2, ny + NODE_SIZE - 2, fill);

            // Локализованная первая буква
            String name = Component.translatable(n.nameKey()).getString();
            String label = name.isEmpty() ? "?" : name.substring(0, 1).toUpperCase();
            
            int tw = font.width(label);
            gfx.drawString(font, label, nx + NODE_SIZE / 2 - tw / 2, ny + NODE_SIZE / 2 - 4, 0xFFFFFFFF, false);
        }

        gfx.pose().popPose();

        String ptsText = Component.translatable("gui.grimoiremod.skill_tree.points", ptsStr).getString();
        gfx.drawString(font, ptsText, 10, 30, 0xFFFFFFFF, true);

        if (hoverNodeId != null) {
            SkillTreeDef.Node n = SkillTreeDef.byId(hoverNodeId);
            if (n != null) {
                List<Component> tooltip = new ArrayList<>();
                tooltip.add(Component.translatable(n.nameKey()).withStyle(ChatFormatting.YELLOW));

                if (!hasBook) {
                    tooltip.add(Component.translatable("gui.grimoiremod.skill_tree.requires_grimoire").withStyle(ChatFormatting.RED));
                } else if (SkillTreeData.isUnlocked(book, n.id())) {
                    int level = SkillTreeData.getNodeLevel(book, n.id());
                    float exp = SkillTreeData.getNodeExp(book, n.id());
                    tooltip.add(Component.translatable("gui.grimoiremod.skill_tree.level", level, n.maxLevel()));
                    if (level < n.maxLevel()) {

                        if (ptsFrac >= 1.0f) {
                            tooltip.add(Component.translatable("gui.grimoiremod.skill_tree.level_up").withStyle(ChatFormatting.GREEN));
                        }
                    } else {
                        tooltip.add(Component.translatable("gui.grimoiremod.skill_tree.max_level").withStyle(ChatFormatting.GOLD));
                    }
                } else {
                    List<String> missingPrereqs = new ArrayList<>();
                    for (String p : n.prereqs()) {
                        if (!SkillTreeData.isUnlocked(book, p)) {
                            SkillTreeDef.Node pNode = SkillTreeDef.byId(p);
                            missingPrereqs.add(pNode != null ? Component.translatable(pNode.nameKey()).getString() : p);
                        }
                    }
                    
                    if (!missingPrereqs.isEmpty()) {
                        tooltip.add(Component.translatable("gui.grimoiremod.skill_tree.requires_node", String.join(", ", missingPrereqs)).withStyle(ChatFormatting.RED));
                    } else if (ptsFrac < n.cost()) {
                        tooltip.add(Component.translatable("gui.grimoiremod.skill_tree.no_points").withStyle(ChatFormatting.RED));
                    } else {
                        tooltip.add(Component.translatable("gui.grimoiremod.skill_tree.unlock", n.cost()).withStyle(ChatFormatting.GREEN));
                    }
                }
                gfx.renderTooltip(font, tooltip, java.util.Optional.empty(), mx, my);
            }
        }

        if (selectedNodeId != null) {
            SkillTreeDef.Node n = SkillTreeDef.byId(selectedNodeId);
            if (n != null) {
                renderComingSoon(gfx, n, hasBook ? book : ItemStack.EMPTY);
            }
        }
    }

    private void renderTabs(GuiGraphics gfx, int mx, int my) {
        String[] labels = {
                Component.translatable("gui.grimoiremod.skill_tree.general").getString(),
                Component.translatable("gui.grimoiremod.skill_tree.main").getString()
        };
        int totalW = 0;
        for (String l : labels) totalW += font.width(l) + 16 + 2;
        int x = width / 2 - totalW / 2;
        int y = 6;
        for (int t = 0; t < 2; t++) {
            int w = font.width(labels[t]) + 16;
            boolean active = t == activeTab;
            boolean hover = inRect(mx, my, x, y, w, 20);
            gfx.fill(x, y, x + w, y + 20, active ? 0xEEFFFFFF : (hover ? 0x44FFFFFF : 0x88000000));
            gfx.drawString(font, labels[t], x + 8, y + 6, active ? 0xFF000000 : 0xFFAAAAAA, false);
            if (active) gfx.fill(x, y + 20, x + w, y + 22, 0xFFFFAA00);
            tabRects[t] = new int[]{x, y, w, 20};
            x += w + 2;
        }
    }

    private void renderComingSoon(GuiGraphics gfx, SkillTreeDef.Node n, ItemStack book) {
        int w = 200, h = 120;
        int px = (width - w) / 2;
        int py = (height - h) / 2;
        
        gfx.fill(0, 0, width, height, 0x88000000);

        gfx.fill(px + 2, py + 2, px + w - 2, py + h - 2, 0xFFF2E8CF);

        gfx.drawString(font, Component.translatable(n.nameKey()).getString(), px + 10, py + 10, 0xFF3A2E20, false);
        gfx.fill(px + 10, py + 22, px + w - 10, py + 23, 0xFF886A4A);

        int level = book.isEmpty() ? 0 : SkillTreeData.getNodeLevel(book, n.id());
        float exp = book.isEmpty() ? 0 : SkillTreeData.getNodeExp(book, n.id());
        
        gfx.drawString(font, "Level: " + level + " / " + n.maxLevel(), px + 10, py + 35, 0xFF3A2E20, false);
        gfx.drawString(font, String.format("Progress: %.1f%%", exp * 100), px + 10, py + 50, 0xFF3A2E20, false);
        
        gfx.drawString(font, "Coming soon...", px + 10, py + 80, 0xFFAA4444, false);
        gfx.drawString(font, "Click outside to close", px + 10, py + h - 15, 0xFF888888, false);
    }

    private void drawLine(GuiGraphics gfx, int x1, int y1, int x2, int y2, int color) {
        int dx = Math.abs(x2 - x1), dy = Math.abs(y2 - y1);
        int sx = x1 < x2 ? 1 : -1, sy = y1 < y2 ? 1 : -1;
        int err = dx - dy;
        while (true) {
            gfx.fill(x1 - 2, y1 - 2, x1 + 2, y1 + 2, color);
            if (x1 == x2 && y1 == y2) break;
            int e2 = 2 * err;
            if (e2 > -dy) { err -= dy; x1 += sx; }
            if (e2 < dx) { err += dx; y1 += sy; }
        }
    }

    @Override
    public boolean mouseClicked(double mx, double my, int button) {
        for (int t = 0; t < 2; t++) {
            int[] r = tabRects[t];
            if (r[2] > 0 && inRect(mx, my, r[0], r[1], r[2], r[3])) {
                activeTab = t;
                selectedNodeId = null;
                return true;
            }
        }

        if (activeTab == TAB_GENERAL) return super.mouseClicked(mx, my, button);

        if (selectedNodeId != null) {
            selectedNodeId = null;
            return true;
        }

        double worldMx = (mx - (width / 2.0f + panX)) / zoom;
        double worldMy = (my - (height / 2.0f + panY)) / zoom;

        for (SkillTreeDef.Node n : SkillTreeDef.NODES) {
            int nx = n.x() * SPACING - NODE_SIZE / 2;
            int ny = n.y() * SPACING - NODE_SIZE / 2;
            if (worldMx >= nx && worldMx < nx + NODE_SIZE && worldMy >= ny && worldMy < ny + NODE_SIZE) {
                ItemStack book = book();
                if (book.isEmpty()) return true;

                if (button == 0) {
                    if (!SkillTreeData.isUnlocked(book, n.id())) {
                        boolean hasPrereqs = true;
                        for (String p : n.prereqs()) {
                            if (!SkillTreeData.isUnlocked(book, p)) { hasPrereqs = false; break; }
                        }
                        if (hasPrereqs && SkillTreeData.getPointsFrac(book) >= n.cost()) {
                            PacketDistributor.sendToServer(UnlockSkillNodePacket.of(itemSlotId, n.id()));
                        }
                    } else {
                        selectedNodeId = n.id();
                    }
                } else if (button == 1) {
                    if (SkillTreeData.isUnlocked(book, n.id())) {
                        int level = SkillTreeData.getNodeLevel(book, n.id());
                        if (level < n.maxLevel() && SkillTreeData.getPointsFrac(book) >= 1.0f) {
                            PacketDistributor.sendToServer(new LevelUpSkillNodePacket(itemSlotId, n.id()));
                        }
                    }
                }
                return true;
            }
        }

        if (button == 0) {
            isDragging = true;
            lastMouseX = mx;
            lastMouseY = my;
            return true;
        }
        return super.mouseClicked(mx, my, button);
    }

    @Override
    public boolean mouseReleased(double mx, double my, int button) {
        isDragging = false;
        return super.mouseReleased(mx, my, button);
    }

    @Override
    public boolean mouseDragged(double mx, double my, int button, double dx, double dy) {
        if (isDragging) {
            panX += (float) (mx - lastMouseX);
            panY += (float) (my - lastMouseY);
            lastMouseX = mx;
            lastMouseY = my;
            return true;
        }
        return super.mouseDragged(mx, my, button, dx, dy);
    }

    @Override
    public boolean mouseScrolled(double mx, double my, double dx, double dy) {
        if (activeTab == TAB_GENERAL) return super.mouseScrolled(mx, my, dx, dy);
        float oldZoom = zoom;
        zoom = (float) Math.max(0.4, Math.min(2.0, zoom + Math.signum(dy) * 0.1));
        float scaleDiff = zoom - oldZoom;
        panX -= (float) (mx - width / 2.0f) * scaleDiff / oldZoom;
        panY -= (float) (my - height / 2.0f) * scaleDiff / oldZoom;
        return true;
    }

    @Override
    public boolean keyPressed(int keyCode, int scanCode, int modifiers) {
        if (keyCode == GLFW.GLFW_KEY_ESCAPE || keyCode == GLFW.GLFW_KEY_K) { onClose(); return true; }
        return super.keyPressed(keyCode, scanCode, modifiers);
    }

    private static boolean inRect(double mx, double my, int x, int y, int w, int h) {
        return mx >= x && mx < x + w && my >= y && my < y + h;
    }
}

package com.example.grimoiremod.data;

import com.example.grimoiremod.glyphs.Stat;
import net.minecraft.nbt.CompoundTag;

import java.util.HashMap;
import java.util.Map;
import java.util.Objects;

public final class GlyphInstance {
    private final String typeId;
    private final Map<Stat, Integer> allocations;

    public GlyphInstance(String typeId) {
        this(typeId, new HashMap<>());
    }

    public GlyphInstance(String typeId, Map<Stat, Integer> allocations) {
        this.typeId = typeId;
        this.allocations = new HashMap<>(allocations);
    }

    public String typeId() { return typeId; }
    public Map<Stat, Integer> allocations() { return allocations; }

    public int getPoints(Stat stat) {
        return allocations.getOrDefault(stat, 0);
    }

    public int totalPoints() {
        return allocations.values().stream().mapToInt(Integer::intValue).sum();
    }

    public CompoundTag toNbt() {
        CompoundTag t = new CompoundTag();
        t.putString("Type", typeId);
        CompoundTag allocs = new CompoundTag();
        allocations.forEach((stat, val) -> allocs.putInt(stat.name(), val));
        t.put("Allocs", allocs);
        return t;
    }

    public static GlyphInstance fromNbt(CompoundTag t) {
        String type = t.getString("Type");
        Map<Stat, Integer> allocs = new HashMap<>();
        CompoundTag a = t.getCompound("Allocs");
        for (String key : a.getAllKeys()) {
            try { allocs.put(Stat.valueOf(key), a.getInt(key)); }
            catch (IllegalArgumentException ignored) {}
        }
        return new GlyphInstance(type, allocs);
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        GlyphInstance that = (GlyphInstance) o;
        return typeId.equals(that.typeId) && allocations.equals(that.allocations);
    }

    @Override
    public int hashCode() {
        return Objects.hash(typeId, allocations);
    }
}

package com.example.grimoiremod.data;

import net.minecraft.nbt.*;

public final class GlyphPage {
    // bump при изменении формата GlyphPage; см. GrimoireData.readRoot
    public static final int VERSION = 12;

    public enum Kind { SPELL, TOC }

    public Kind kind;
    public int slotCount;
    public GlyphInstance[] slots;
    public String name;
    public int bookmarkColor;
    public boolean favorite;

    public GlyphPage(int slotCount, int bookmarkColor) {
        this(Kind.SPELL, slotCount, bookmarkColor);
    }

    public GlyphPage(Kind kind, int slotCount, int bookmarkColor) {
        this.kind = kind;
        this.slotCount = (kind == Kind.TOC) ? 0 : Math.max(1, slotCount);
        this.slots = new GlyphInstance[this.slotCount];
        this.name = "";
        this.bookmarkColor = bookmarkColor;
        this.favorite = (kind == Kind.TOC); // TOC всегда избранное
    }

    public static GlyphPage toc(int bookmarkColor) {
        return new GlyphPage(Kind.TOC, 0, bookmarkColor);
    }

    public boolean isSpell() { return kind == Kind.SPELL; }
    public boolean isToc()   { return kind == Kind.TOC; }

    public CompoundTag toNbt() {
        CompoundTag t = new CompoundTag();
        t.putString("Kind", kind.name());
        t.putInt("SlotCount", slotCount);
        ListTag list = new ListTag();
        for (GlyphInstance inst : slots) list.add(inst == null ? new CompoundTag() : inst.toNbt());
        t.put("Slots", list);
        t.putString("Name", name == null ? "" : name);
        t.putInt("Color", bookmarkColor);
        t.putBoolean("Favorite", favorite);
        return t;
    }

    public static GlyphPage fromNbt(CompoundTag t) {
        Kind kind = Kind.TOC.name().equals(t.getString("Kind")) ? Kind.TOC : Kind.SPELL;
        int n = t.getInt("SlotCount");
        if (kind == Kind.SPELL && n <= 0) n = 7;
        GlyphPage p = new GlyphPage(kind, n, t.getInt("Color"));
        p.name = t.getString("Name");
        p.favorite = (kind == Kind.TOC) || t.getBoolean("Favorite");

        if (t.contains("Slots")) {
            Tag slotsTag = t.get("Slots");
            if (slotsTag instanceof ListTag list) {
                for (int i = 0; i < p.slotCount && i < list.size(); i++) {
                    Tag elem = list.get(i);
                    if (elem instanceof StringTag sTag) {
                        if (!sTag.getAsString().isEmpty()) p.slots[i] = new GlyphInstance(sTag.getAsString());
                    } else if (elem instanceof CompoundTag cTag) {
                        if (!cTag.isEmpty() && cTag.contains("Type")) p.slots[i] = GlyphInstance.fromNbt(cTag);
                    }
                }
            }
        }
        return p;
    }
}

import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.function.Consumer;

public final class GrimoireData {
    private static final String ROOT = "Grimoire";
    private static final String VERSION_KEY = "Version";
    private static final String ACTIVE_PAGE = "ActivePage";
    private static final String PAGES = "Pages";
    private static final String BANK = "Bank";

    private static final int MAX_POINTS_PER_GLYPH = 1000;
    private static final int MAX_NAME_LEN = 32;

    private static final int[] BOOKMARK_PALETTE = {
            0xFFC9A64A, 0xFFCC4444, 0xFF44CC44, 0xFF4444CC, 0xFFAA44AA,
            0xFF44AAAA, 0xFFAAAA44, 0xFFCC7733, 0xFF7733CC, 0xFF33CC77
    };

    private GrimoireData() {}

    private static int paletteColor(int index) {
        return BOOKMARK_PALETTE[Math.floorMod(index, BOOKMARK_PALETTE.length)];
    }

    static CompoundTag copyRoot(ItemStack stack) {
        CustomData cd = stack.get(DataComponents.CUSTOM_DATA);
        return cd != null ? cd.copyTag() : new CompoundTag();
    }
    static void saveRoot(ItemStack stack, CompoundTag root) {
        stack.set(DataComponents.CUSTOM_DATA, CustomData.of(root));
    }

    static CompoundTag readRoot(ItemStack stack) {
        CompoundTag root = copyRoot(stack);
        boolean needFresh;
        if (!root.contains(ROOT)) needFresh = true;
        else needFresh = root.getCompound(ROOT).getInt(VERSION_KEY) != GlyphPage.VERSION;
        if (needFresh) {
            CompoundTag fresh = (stack.getItem() instanceof CreativeGrimoireItem)
                    ? freshCreativeGrimoire() : freshGrimoire();
            if (root.contains(ROOT)) {
                tryMigrate(root.getCompound(ROOT), fresh);
            }
            root.put(ROOT, fresh);
            saveRoot(stack, root);
        }
        return root;
    }

    private static void tryMigrate(CompoundTag old, CompoundTag fresh) {
        try {
            if (old.contains(PAGES)) {
                ListTag oldPages = old.getList(PAGES, Tag.TAG_COMPOUND);
                ListTag newPages = new ListTag();
                newPages.add(GlyphPage.toc(paletteColor(0)).toNbt());
                int limit = Math.min(oldPages.size(), 30);
                for (int i = 0; i < limit; i++) {
                    try {
                        GlyphPage gp = GlyphPage.fromNbt(oldPages.getCompound(i));
                        if (gp.isToc()) continue;
                        newPages.add(gp.toNbt());
                    } catch (Exception ignored) {}
                }
                if (newPages.size() > 1) fresh.put(PAGES, newPages);
            }
            if (old.contains(BANK)) {
                ListTag oldBank = old.getList(BANK, Tag.TAG_COMPOUND);
                ListTag newBank = new ListTag();
                int limit = Math.min(oldBank.size(), 500);
                for (int i = 0; i < limit; i++) {
                    try {
                        GlyphInstance inst = GlyphInstance.fromNbt(oldBank.getCompound(i));
                        if (Glyphs.byId(inst.typeId()) != null) newBank.add(inst.toNbt());
                    } catch (Exception ignored) {}
                }
                if (!newBank.isEmpty()) fresh.put(BANK, newBank);
            }
            if (old.contains(SkillTreeData.KEY)) {
                fresh.put(SkillTreeData.KEY, old.getCompound(SkillTreeData.KEY).copy());
            }
            if (old.contains(ACTIVE_PAGE)) {
                fresh.putInt(ACTIVE_PAGE, old.getInt(ACTIVE_PAGE));
            }
        } catch (Exception ignored) {}
    }

    private static void withPage(ItemStack stack, int index, Consumer<GlyphPage> fn) {
        CompoundTag root = readRoot(stack);
        CompoundTag g = root.getCompound(ROOT);
        ListTag pages = g.getList(PAGES, Tag.TAG_COMPOUND);
        if (index < 0 || index >= pages.size()) return;
        GlyphPage page = GlyphPage.fromNbt(pages.getCompound(index));
        fn.accept(page);
        ListTag newList = new ListTag();
        for (int i = 0; i < pages.size(); i++)
            newList.add(i == index ? page.toNbt() : pages.getCompound(i).copy());
        g.put(PAGES, newList);
        saveRoot(stack, root);
    }

        if (count == 0) return;

    }

    public static void switchPage(ItemStack stack, boolean next) {
        int count = getPageCount(stack);
        if (count <= 1) return;
        int active = getActivePage(stack);
        int dir = next ? 1 : -1;
        for (int i = 0; i < count; i++) {
            active = ((active + dir) % count + count) % count;
            GlyphPage p = getPage(stack, active);
            if (p != null && !p.isToc()) { setActivePage(stack, active); return; }
        }
    }

    public static boolean placeGlyphWithAllocs(ItemStack stack, int pageIdx, int slot, String typeId, Map<Stat, Integer> allocs) {
        Glyph g = Glyphs.byId(typeId);
        if (g == null) return false;

        for (var entry : allocs.entrySet()) {
            int v = entry.getValue();
            if (v < 0 || v > MAX_POINTS_PER_GLYPH) return false;
        }

        CompoundTag root = readRoot(stack);
        CompoundTag gr = root.getCompound(ROOT);
        ListTag pages = gr.getList(PAGES, Tag.TAG_COMPOUND);
        if (pageIdx < 0 || pageIdx >= pages.size()) return false;
        GlyphPage page = GlyphPage.fromNbt(pages.getCompound(pageIdx));
        if (!page.isSpell() || slot < 0 || slot >= page.slotCount || page.slots[slot] != null) return false;

        ListTag bank = gr.getList(BANK, Tag.TAG_COMPOUND);
        GlyphInstance toPlace = null;
        for (int i = 0; i < bank.size(); i++) {
            GlyphInstance inst = GlyphInstance.fromNbt(bank.getCompound(i));
            if (inst.typeId().equals(typeId) && inst.allocations().equals(allocs)) {
                toPlace = inst;
                bank.remove(i);
                break;
            }
        }
        if (toPlace == null) return false;

        page.slots[slot] = toPlace;
        ListTag newList = new ListTag();
        for (int i = 0; i < pages.size(); i++)
            newList.add(i == pageIdx ? page.toNbt() : pages.getCompound(i).copy());
        gr.put(PAGES, newList);
        gr.put(BANK, bank);
        saveRoot(stack, root);
        return true;
    }

    public static boolean moveGlyph(ItemStack stack, int pageIdx, int fromSlot, int toSlot) {
        if (fromSlot == toSlot) return false;
        CompoundTag root = readRoot(stack);
        CompoundTag gr = root.getCompound(ROOT);
        ListTag pages = gr.getList(PAGES, Tag.TAG_COMPOUND);
        if (pageIdx < 0 || pageIdx >= pages.size()) return false;
        GlyphPage page = GlyphPage.fromNbt(pages.getCompound(pageIdx));
        if (!page.isSpell()) return false;
        if (fromSlot < 0 || fromSlot >= page.slotCount || toSlot < 0 || toSlot >= page.slotCount) return false;
        GlyphInstance from = page.slots[fromSlot];
        if (from == null) return false;
        if (page.slots[toSlot] != null) return false;

        page.slots[toSlot] = from;
        page.slots[fromSlot] = null;
        ListTag newList = new ListTag();
        for (int i = 0; i < pages.size(); i++)
            newList.add(i == pageIdx ? page.toNbt() : pages.getCompound(i).copy());
        gr.put(PAGES, newList);
        saveRoot(stack, root);
        return true;
    }

    public static boolean removeGlyph(ItemStack stack, int pageIdx, int slot) {
        CompoundTag root = readRoot(stack);
        CompoundTag gr = root.getCompound(ROOT);
        ListTag pages = gr.getList(PAGES, Tag.TAG_COMPOUND);
        if (pageIdx < 0 || pageIdx >= pages.size()) return false;
        GlyphPage page = GlyphPage.fromNbt(pages.getCompound(pageIdx));
        if (!page.isSpell() || slot < 0 || slot >= page.slotCount) return false;
        GlyphInstance removed = page.slots[slot];
        if (removed == null) return false;

        ListTag bank = gr.getList(BANK, Tag.TAG_COMPOUND);
        bank.add(removed.toNbt());

        page.slots[slot] = null;
        ListTag newList = new ListTag();
        for (int i = 0; i < pages.size(); i++)
            newList.add(i == pageIdx ? page.toNbt() : pages.getCompound(i).copy());
        gr.put(PAGES, newList);
        gr.put(BANK, bank);
        saveRoot(stack, root);
        return true;
    }

    public static void movePage(ItemStack stack, int from, int to) {
        CompoundTag root = readRoot(stack);
        CompoundTag g = root.getCompound(ROOT);
        ListTag pages = g.getList(PAGES, Tag.TAG_COMPOUND);
        if (from < 0 || from >= pages.size() || to < 0 || to >= pages.size() || from == to) return;
        if (GlyphPage.fromNbt(pages.getCompound(from)).isToc()) return;
        if (GlyphPage.fromNbt(pages.getCompound(to)).isToc()) return;

        List<CompoundTag> list = new ArrayList<>();
        for (int i = 0; i < pages.size(); i++) list.add(pages.getCompound(i).copy());
        CompoundTag moved = list.remove(from);
        list.add(to, moved);
        ListTag newList = new ListTag();
        for (CompoundTag c : list) newList.add(c);
        g.put(PAGES, newList);

        int active = g.getInt(ACTIVE_PAGE);
        if (active == from) active = to;
        else if (from < active && to >= active) active--;
        else if (from > active && to <= active) active++;
        g.putInt(ACTIVE_PAGE, active);
        saveRoot(stack, root);
    }

    public static int getStatBudget(ItemStack stack, Stat stat) {
        String nodeId = switch (stat) {
            case DAMAGE -> "grimoiremod:combat_damage";
            case HEAL -> "grimoiremod:support_heal";
            case COOLDOWN -> "grimoiremod:utility_cooldown";
            case ANGLE -> "grimoiremod:combat_radius";
            default -> null;
        };
        if (nodeId == null) return 0; // Если стат не прокачивается, бюджет 0
        return SkillTreeData.getNodeLevel(stack, nodeId);
    }

    public static int getSpentPoints(ItemStack stack, Stat stat) {
        CompoundTag gr = readRoot(stack).getCompound(ROOT);
        int spent = 0;
        ListTag pages = gr.getList(PAGES, Tag.TAG_COMPOUND);
        for (int i = 0; i < pages.size(); i++) {
            GlyphPage page = GlyphPage.fromNbt(pages.getCompound(i));
            if (page.isToc()) continue;
            for (GlyphInstance inst : page.slots) if (inst != null) spent += inst.getPoints(stat);
        }
        ListTag bank = gr.getList(BANK, Tag.TAG_COMPOUND);
        for (int i = 0; i < bank.size(); i++) spent += GlyphInstance.fromNbt(bank.getCompound(i)).getPoints(stat);
        return spent;
    }

    public static boolean canAllocate(ItemStack stack, Stat stat, int delta) {
        if (delta <= 0) return true;
        return getSpentPoints(stack, stat) + delta <= getStatBudget(stack, stat);
    }

    public static void allocatePoint(ItemStack stack, int pageIdx, int slotIdx, Stat stat, int delta) {
        if (delta > 0 && !canAllocate(stack, stat, delta)) return;
        withPage(stack, pageIdx, page -> {
            if (slotIdx < 0 || slotIdx >= page.slotCount) return;
            GlyphInstance inst = page.slots[slotIdx];
            if (inst == null) return;
            Glyph g = Glyphs.byId(inst.typeId());
            if (g != null && g.advanced()) return; // Развитые нельзя
            Map<Stat, Integer> newAllocs = new HashMap<>(inst.allocations());
            int current = newAllocs.getOrDefault(stat, 0);
            int next = Math.max(0, current + delta);
            if (next == current) return;
            if (next > MAX_POINTS_PER_GLYPH) return;
            if (next == 0) newAllocs.remove(stat); else newAllocs.put(stat, next);
            page.slots[slotIdx] = new GlyphInstance(inst.typeId(), newAllocs);
        });
    }

    public static void allocatePointToBank(ItemStack stack, int bankIndex, Stat stat, int delta) {
        if (delta > 0 && !canAllocate(stack, stat, delta)) return;
        CompoundTag root = readRoot(stack);
        CompoundTag gr = root.getCompound(ROOT);
        ListTag bank = gr.getList(BANK, Tag.TAG_COMPOUND);
        if (bankIndex < 0 || bankIndex >= bank.size()) return;

        GlyphInstance inst = GlyphInstance.fromNbt(bank.getCompound(bankIndex));
        Glyph g = Glyphs.byId(inst.typeId());
        if (g != null && g.advanced()) return;
        Map<Stat, Integer> newAllocs = new HashMap<>(inst.allocations());
        int current = newAllocs.getOrDefault(stat, 0);
        int next = Math.max(0, current + delta);
        if (next == current) return;
        if (next > MAX_POINTS_PER_GLYPH) return;
        if (next == 0) newAllocs.remove(stat); else newAllocs.put(stat, next);

        bank.set(bankIndex, new GlyphInstance(inst.typeId(), newAllocs).toNbt());
        gr.put(BANK, bank);
        saveRoot(stack, root);
    }

    public static void resetGlyph(ItemStack stack, int pageIdx, int slotIdx) {
        withPage(stack, pageIdx, page -> {
            if (slotIdx < 0 || slotIdx >= page.slotCount) return;
            GlyphInstance inst = page.slots[slotIdx];
            if (inst != null) page.slots[slotIdx] = new GlyphInstance(inst.typeId());
        });
    }

    public static void resetGlyphInBank(ItemStack stack, int bankIndex) {
        CompoundTag root = readRoot(stack);
        CompoundTag gr = root.getCompound(ROOT);
        ListTag bank = gr.getList(BANK, Tag.TAG_COMPOUND);
        if (bankIndex < 0 || bankIndex >= bank.size()) return;
        GlyphInstance inst = GlyphInstance.fromNbt(bank.getCompound(bankIndex));
        bank.set(bankIndex, new GlyphInstance(inst.typeId()).toNbt());
        gr.put(BANK, bank);
        saveRoot(stack, root);
    }

package com.example.grimoiremod.data;

import net.minecraft.nbt.CompoundTag;
import net.minecraft.nbt.ListTag;
import net.minecraft.nbt.StringTag;
import net.minecraft.nbt.Tag;
import net.minecraft.world.item.ItemStack;

import java.util.ArrayList;
import java.util.List;

public final class SkillTreeData {
    public static final String KEY      = "SkillTree";
    public static final String POINTS   = "Points";
    public static final String UNLOCKED = "Unlocked";
    public static final String LEVELS   = "Levels";
    public static final String EXP      = "Exp";

    private static final String ROOT = "Grimoire";
    public static final int STARTING_POINTS = 5;

    private SkillTreeData() {}

    public static CompoundTag fresh() {
        CompoundTag t = new CompoundTag();
        t.putFloat(POINTS, STARTING_POINTS);
        t.put(UNLOCKED, new ListTag());
        t.put(LEVELS, new CompoundTag());
        t.put(EXP, new CompoundTag());
        return t;
    }

    private static CompoundTag treeOrNull(ItemStack stack) {
        CompoundTag root = GrimoireData.readRoot(stack);
        CompoundTag g = root.getCompound(ROOT);
        return g.contains(KEY) ? g.getCompound(KEY) : null;
    }

    public static float getPointsFrac(ItemStack stack) {
        CompoundTag tree = treeOrNull(stack);
        if (tree == null) return 0f;
        return tree.contains(POINTS, Tag.TAG_FLOAT) ? tree.getFloat(POINTS) : tree.getInt(POINTS);
    }

    public static int getPoints(ItemStack stack) {
        return (int)getPointsFrac(stack);
    }

    public static boolean isUnlocked(ItemStack stack, String nodeId) {
        CompoundTag tree = treeOrNull(stack);
        return tree != null && isUnlockedIn(tree, nodeId);
    }

    private static boolean isUnlockedIn(CompoundTag tree, String nodeId) {
        ListTag list = tree.getList(UNLOCKED, Tag.TAG_STRING);
        for (int i = 0; i < list.size(); i++) {
            if (list.getString(i).equals(nodeId)) return true;
        }
        return false;
    }

    public static int getNodeLevel(ItemStack stack, String nodeId) {
        CompoundTag tree = treeOrNull(stack);
        if (tree == null) return 0;
        return tree.getCompound(LEVELS).getInt(nodeId);
    }

    public static float getNodeExp(ItemStack stack, String nodeId) {
        CompoundTag tree = treeOrNull(stack);
        if (tree == null) return 0;
        return tree.getCompound(EXP).getFloat(nodeId);
    }

    public static void addNodeLevel(ItemStack stack, String nodeId, int amount) {
        CompoundTag root = GrimoireData.readRoot(stack);
        CompoundTag g = root.getCompound(ROOT);
        CompoundTag tree = g.contains(KEY) ? g.getCompound(KEY) : fresh();
        CompoundTag levels = tree.getCompound(LEVELS);
        levels.putInt(nodeId, levels.getInt(nodeId) + amount);
        tree.put(LEVELS, levels);
        g.put(KEY, tree);
        root.put(ROOT, g);
        GrimoireData.saveRoot(stack, root);
    }

    public static void addNodeExp(ItemStack stack, String nodeId, float amount, int maxLevel) {
        CompoundTag root = GrimoireData.readRoot(stack);
        CompoundTag g = root.getCompound(ROOT);
        CompoundTag tree = g.contains(KEY) ? g.getCompound(KEY) : fresh();

        CompoundTag levels = tree.getCompound(LEVELS);
        CompoundTag exp = tree.getCompound(EXP);

        int currentLevel = levels.getInt(nodeId);
        if (currentLevel >= maxLevel) return;

        float currentExp = exp.getFloat(nodeId) + amount;
        while (currentExp >= 1.0f && currentLevel < maxLevel) {
            currentExp -= 1.0f;
            currentLevel++;
        }
        levels.putInt(nodeId, currentLevel);
        exp.putFloat(nodeId, currentExp);

        tree.put(LEVELS, levels);
        tree.put(EXP, exp);
        g.put(KEY, tree);
        root.put(ROOT, g);
        GrimoireData.saveRoot(stack, root);
    }

    public static int addPoints(ItemStack stack, float amount) {
        CompoundTag root = GrimoireData.readRoot(stack);
        CompoundTag g = root.getCompound(ROOT);
        CompoundTag tree = g.contains(KEY) ? g.getCompound(KEY) : fresh();

        float cur = tree.contains(POINTS, Tag.TAG_FLOAT) ? tree.getFloat(POINTS) : tree.getInt(POINTS);
        float next = Math.max(0, cur + amount);
        tree.putFloat(POINTS, next);

        g.put(KEY, tree);
        root.put(ROOT, g);
        GrimoireData.saveRoot(stack, root);
        return (int)next;
    }

    public static boolean tryUnlock(ItemStack stack, String nodeId, int cost, List<String> prereqs) {
        CompoundTag root = GrimoireData.readRoot(stack);
        CompoundTag g = root.getCompound(ROOT);
        CompoundTag tree = g.contains(KEY) ? g.getCompound(KEY) : fresh();

        if (isUnlockedIn(tree, nodeId)) return false;
        for (String p : prereqs) if (!isUnlockedIn(tree, p)) return false;
        
        float pts = tree.contains(POINTS, Tag.TAG_FLOAT) ? tree.getFloat(POINTS) : tree.getInt(POINTS);
        if (pts < cost) return false;

        tree.putFloat(POINTS, pts - cost);
        ListTag list = tree.getList(UNLOCKED, Tag.TAG_STRING);
        ListTag newList = new ListTag();
        for (int i = 0; i < list.size(); i++) newList.add(list.get(i));
        newList.add(StringTag.valueOf(nodeId));
        tree.put(UNLOCKED, newList);
