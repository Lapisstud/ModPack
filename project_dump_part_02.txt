        // При разблокировке сразу ставим 1 уровень
        CompoundTag levels = tree.getCompound(LEVELS);
        if (levels.getInt(nodeId) == 0) {
            levels.putInt(nodeId, 1);
            tree.put(LEVELS, levels);
        }

        g.put(KEY, tree);
        root.put(ROOT, g);
        GrimoireData.saveRoot(stack, root);
        return true;
    }

public final class SpellDescriber {
    private SpellDescriber() {}

    public static List<Component> describeLines(GlyphPage page) {
        List<Component> out = new ArrayList<>();
        if (page == null) return out;
        if (page.isToc()) {
            out.add(Component.translatable("gui.grimoiremod.toc").withStyle(ChatFormatting.BOLD));
            return out;
        }

        SpellParser.Parsed parsed = SpellParser.parse(page);
        if (parsed.formId() == null) {
            out.add(Component.translatable("gui.grimoiremod.no_form").withStyle(ChatFormatting.RED));
            if (parsed.orphanModifier()) {
                out.add(Component.translatable("gui.grimoiremod.orphan_modifier").withStyle(ChatFormatting.RED));
            }
        } else {
            StatBlock sb = SpellAggregator.aggregate(page);

            for (SpellParser.Element e : parsed.elements()) {
                Glyph g = Glyphs.byId(e.instance().typeId());
                String name = g != null ? g.getDisplayName().getString() : e.instance().typeId();
                if (g != null && g.advanced()) {
                    name = Component.translatable("gui.grimoiremod.advanced", name).getString();
                }

                double baseDmg  = g != null ? g.baseStat(Stat.DAMAGE) : 0;
                double baseHeal = g != null ? g.baseStat(Stat.HEAL)   : 0;

                if (baseDmg > 0) {
                    double total = SpellAggregator.elementBaseDamage(e, sb);
                    ChatFormatting sign = total > 0 ? ChatFormatting.GREEN : (total < 0 ? ChatFormatting.RED : ChatFormatting.WHITE);
                    int col = g.color() & 0xFFFFFF;
                    out.add(Component.literal(name + ": ")
                            .withStyle(style -> style.withColor(col))
                            .append(Component.translatable("gui.grimoiremod.damage", Stat.trimNumber(total)).withStyle(sign)));
                } else if (baseHeal > 0) {
                    double total = SpellAggregator.elementBaseHeal(e, sb);
                    int col = g.color() & 0xFFFFFF;
                    out.add(Component.literal(name + ": ")
                            .withStyle(style -> style.withColor(col))
                            .append(Component.translatable("gui.grimoiremod.heal", Stat.trimNumber(total)).withStyle(ChatFormatting.GREEN)));
                } else {
                    int col = g != null ? (g.color() & 0xFFFFFF) : 0xFFFFFF;
                    out.add(Component.literal(name).withStyle(style -> style.withColor(col)));
                }
            }

            if (!parsed.elements().isEmpty()) out.add(Component.literal(" "));
            out.add(Component.literal("---").withStyle(ChatFormatting.DARK_GRAY));

            Glyph form = Glyphs.byId(parsed.formId());
            String formName = form != null ? form.getDisplayName().getString() : parsed.formId();
            int count = sb.getInt(Stat.COUNT);
            out.add(Component.translatable("gui.grimoiremod.form", formName + (count > 1 ? " x" + count : ""))
                    .withStyle(ChatFormatting.BOLD));

            if (parsed.formId().equals("projectile")) {
                double critChance = sb.get(Stat.CRIT_CHANCE);
                if (critChance > 0) {
                    out.add(Component.translatable("gui.grimoiremod.crit_chance", (int)Math.round(critChance * 100))
                            .withStyle(ChatFormatting.GOLD));
                }
                double critMult = sb.get(Stat.CRIT_MULT);
                if (critMult > 0) {
                    out.add(Component.translatable("gui.grimoiremod.crit_damage", (int)Math.round(critMult * 100))
                            .withStyle(ChatFormatting.GOLD));
                }
            }

            double cooldown = sb.get(Stat.COOLDOWN);
            if (cooldown > 0) {
                out.add(Component.translatable("gui.grimoiremod.cooldown", Stat.trimNumber(cooldown))
                        .withStyle(ChatFormatting.GRAY));
            }
        }

        if (!parsed.errors().isEmpty()) {
            out.add(Component.literal("---").withStyle(ChatFormatting.DARK_GRAY));
            out.add(Component.translatable("gui.grimoiremod.errors").withStyle(ChatFormatting.RED));
            for (var e : parsed.errors().entrySet()) {
                out.add(Component.literal(" . ").append(Component.translatable(e.getValue())).withStyle(ChatFormatting.RED));
            }
        }

        return out;
    }
}

package com.example.grimoiremod.data;

import com.example.grimoiremod.glyphs.Glyph;
import com.example.grimoiremod.glyphs.GlyphCategory;
import com.example.grimoiremod.glyphs.Glyphs;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public final class SpellParser {
    private SpellParser() {}

    public record Element(GlyphInstance instance, List<String> mods) {}

    public record Parsed(String formId, List<String> formMods, List<Element> elements,
                         boolean orphanModifier, Map<Integer, String> errors) {}

    public static Parsed parse(GlyphPage page) {
        if (page == null || page.isToc()) {
            return new Parsed(null, List.of(), List.of(), false, new HashMap<>());
        }

        String formId = null;
        List<String> formMods = new ArrayList<>();
        List<Element> elements = new ArrayList<>();
        boolean orphanModifier = false;
        List<String> activeAnchorMods = null;
        Map<Integer, String> errors = new HashMap<>();

        for (int i = 0; i < page.slots.length; i++) {
            GlyphInstance inst = page.slots[i];
            if (inst == null) continue;
            Glyph g = Glyphs.byId(inst.typeId());
            if (g == null) continue;

            if (g.category() == GlyphCategory.FORM) {
                if (formId == null) {
                    formId = g.id();
                    activeAnchorMods = formMods;
                    if (g.id().equals("fireball")) {
                        Element e = new Element(inst, new ArrayList<>());
                        elements.add(e);
                        activeAnchorMods = e.mods();
                    }
                } else {
                    errors.put(i, "gui.grimoiremod.error.multiple_forms");
                }
            } else if (g.category() == GlyphCategory.ELEMENT) {
                boolean hasElement = false;
                for (Element e : elements) {
                    Glyph eg = Glyphs.byId(e.instance().typeId());
                    if (eg != null && eg.category() == GlyphCategory.ELEMENT) { hasElement = true; break; }
                }
                if (!hasElement) {
                    Element e = new Element(inst, new ArrayList<>());
                    elements.add(e);
                    activeAnchorMods = e.mods();
                } else {
                    errors.put(i, "gui.grimoiremod.error.multiple_elements");
                }
            } else if (g.category() == GlyphCategory.EFFECT) {
                boolean duplicate = false;
                for (Element e : elements) {
                    if (e.instance().typeId().equals(g.id())) { duplicate = true; break; }
                }
                if (!duplicate) {
                    Element e = new Element(inst, new ArrayList<>());
                    elements.add(e);
                    activeAnchorMods = e.mods();
                } else {
                    errors.put(i, "gui.grimoiremod.error.duplicate_effect");
                }
            } else if (g.category() == GlyphCategory.FORM_MODIFIER) {
                formMods.add(g.id());
            } else if (g.category() == GlyphCategory.MODIFIER) {
                if (activeAnchorMods != null) activeAnchorMods.add(g.id());
                else {
                    orphanModifier = true;
                    errors.put(i, "gui.grimoiremod.error.orphan_modifier");
                }
            }
        }

        if (formId == null && page.slots.length > 0) {
            boolean hasAnyGlyph = false;
            for (GlyphInstance inst : page.slots) if (inst != null) hasAnyGlyph = true;
            if (hasAnyGlyph) errors.put(-1, "gui.grimoiremod.error.no_form");
        }

        return new Parsed(formId, formMods, elements, orphanModifier, errors);
    }
}

    private static final int BASE_MAX_LIFE = 100;

    private static final double SPEED = 1.0;
    private static final int TRAIL_DELAY_TICKS = 3;

    private StatBlock formStats = StatBlock.defaults();

    private float critFactor = 1.0f;

    public SpellProjectile(EntityType<? extends SpellProjectile> type, Level level) {
        super(type, level);
        setNoGravity(true);
    }

    public SpellProjectile(Level level, LivingEntity owner,
                           List<SpellParser.Element> elements,
                           StatBlock formStats, float critFactor, Vec3 direction) {
        super(GrimoireMod.SPELL_PROJECTILE.get(), level);
        setOwner(owner);
        setNoGravity(true);
        this.formStats = formStats;
        this.critFactor = critFactor;
        this.maxLifeMultiplier = formStats.getFloat(Stat.LIFE);
        setElements(elements);
        setPos(owner.getX(), owner.getEyeY() - 0.1, owner.getZ());
        setDeltaMovement(direction.normalize().scale(SPEED));
    }

    @Override
    protected void defineSynchedData(SynchedEntityData.Builder builder) {
        builder.define(PRIMARY, "");
        builder.define(LIFE, 0);
    }

    private int life() { return entityData.get(LIFE); }
    private void setLife(int v) { entityData.set(LIFE, v); }

    public void setElements(List<SpellParser.Element> els) {
        this.elements = new ArrayList<>(els);
        String primary = els.isEmpty() ? "" : els.get(0).instance().typeId();
        entityData.set(PRIMARY, primary);
    }

    public String primaryElement() {
        return entityData.get(PRIMARY);
    }

    @Override
    public void tick() {
        super.tick();

        if (level().isClientSide()) {
            if (life() < TRAIL_DELAY_TICKS) return;
            spawnTrail();
            return;
        }

        setLife(life() + 1);
        int maxLife = Math.max(MIN_MAX_LIFE, (int)(BASE_MAX_LIFE * maxLifeMultiplier));
        if (life() > maxLife) { discard(); return; }

        Vec3 from = position();
        Vec3 to = from.add(getDeltaMovement());

        EntityHitResult entHit = findEntityHit(from, to);
        if (entHit != null) {
            if (entHit.getEntity() instanceof LivingEntity target) {
                for (SpellParser.Element el : elements) {
                    GrimoireEffects.apply(el, formStats, target, critFactor, getOwner(), this);
                }
            }
            burstParticles();
            discard();
            return;
        }

        HitResult blockHit = level().clip(new ClipContext(from, to,
                ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, this));
        if (blockHit.getType() != HitResult.Type.MISS) {
            setPos(blockHit.getLocation());
            burstParticles();
            discard();
            return;
        }

        setPos(to);
    }

    private EntityHitResult findEntityHit(Vec3 from, Vec3 to) {
        EntityHitResult best = null;
        double bestDist = Double.MAX_VALUE;
        AABB sweep = getBoundingBox().expandTowards(getDeltaMovement()).inflate(0.5);
        for (var e : level().getEntities(this, sweep,
                e -> e instanceof LivingEntity && e != getOwner() && e.isAlive())) {
            AABB box = e.getBoundingBox().inflate(0.3);
            var opt = box.clip(from, to);
            if (opt.isPresent()) {
                double d = from.distanceToSqr(opt.get());
                if (d < bestDist) { bestDist = d; best = new EntityHitResult(e, opt.get()); }
            }
        }
        return best;
    }

    private void spawnTrail() {
        Vec3 p = position();
        switch (primaryElement()) {
            case "fire", "fireball" -> {
                level().addParticle(ParticleTypes.FLAME, p.x, p.y, p.z, jitter(0.02), jitter(0.02), jitter(0.02));
                if (random.nextInt(3) == 0)
                    level().addParticle(ParticleTypes.SMOKE, p.x, p.y, p.z, jitter(0.03), 0.02, jitter(0.03));
            }
            case "ice" -> level().addParticle(ParticleTypes.SNOWFLAKE, p.x, p.y, p.z, jitter(0.04), jitter(0.04), jitter(0.04));
            case "water" -> level().addParticle(ParticleTypes.SPLASH, p.x, p.y, p.z, jitter(0.04), jitter(0.04), jitter(0.04));
            case "heal" -> level().addParticle(ParticleTypes.HEART, p.x, p.y, p.z, jitter(0.04), jitter(0.04), jitter(0.04));
            case "speed", "slow" -> level().addParticle(ParticleTypes.ENCHANTED_HIT, p.x, p.y, p.z, jitter(0.04), jitter(0.04), jitter(0.04));
            default -> level().addParticle(ParticleTypes.CRIT, p.x, p.y, p.z, jitter(0.06), jitter(0.06), jitter(0.06));
        }
    }

    private double jitter(double amp) { return (random.nextDouble() - 0.5) * 2 * amp; }

    private void burstParticles() {

    @Override
    protected void readAdditionalSaveData(CompoundTag tag) {

        this.critFactor = tag.contains("CritFactor") ? tag.getFloat("CritFactor") : 1.0f;

        this.formStats = StatBlock.defaults();
        if (tag.contains("FormStats")) {
            CompoundTag fs = tag.getCompound("FormStats");
            EnumMap<Stat, Double> m = new EnumMap<>(Stat.class);
            for (Stat s : Stat.values()) {
                m.put(s, fs.contains(s.name()) ? fs.getDouble(s.name()) : s.defaultValue);
            }
            this.formStats = new StatBlock(m);
        }

        this.elements = new ArrayList<>();
        ListTag list = tag.getList("Elements", Tag.TAG_COMPOUND);
        for (int i = 0; i < list.size(); i++) {
            CompoundTag et = list.getCompound(i);
            GlyphInstance inst = GlyphInstance.fromNbt(et);
            List<String> mods = new ArrayList<>();
            ListTag modList = et.getList("Mods", Tag.TAG_STRING);
            for (int j = 0; j < modList.size(); j++) mods.add(modList.getString(j));
            this.elements.add(new SpellParser.Element(inst, mods));
        }
    }

    @Override
    protected void addAdditionalSaveData(CompoundTag tag) {
        tag.putString("Primary", entityData.get(PRIMARY));
        tag.putInt("Life", life());
        tag.putFloat("LifeMult", maxLifeMultiplier);
        tag.putFloat("CritFactor", critFactor);

        CompoundTag fs = new CompoundTag();
        for (Stat s : Stat.values()) fs.putDouble(s.name(), formStats.get(s));
        tag.put("FormStats", fs);

        ListTag list = new ListTag();
        for (SpellParser.Element e : elements) {
            CompoundTag et = e.instance().toNbt();
            ListTag modList = new ListTag();
            for (String m : e.mods()) modList.add(StringTag.valueOf(m));
            et.put("Mods", modList);
            list.add(et);
        }
        tag.put("Elements", list);
    }

public record Glyph(
        String id,
        GlyphCategory category,
        int color,
        List<StatMod> stats,
        boolean advanced,
        Set<Stat> upgradeableStats
) {
    public Glyph(String id, GlyphCategory category, int color) {
        this(id, category, color, List.of(), false, Set.of());
    }
    public Glyph(String id, GlyphCategory category, int color, List<StatMod> stats) {
        this(id, category, color, stats, false, Set.of());
    }
    public Glyph(String id, GlyphCategory category, int color, List<StatMod> stats, boolean advanced) {
        this(id, category, color, stats, advanced, Set.of());
    }

    public Component getDisplayName() {
        return Component.translatable("glyph.grimoiremod." + id);
    }

    public double baseStat(Stat stat) {
        for (StatMod m : stats) {
            if (m.stat() == stat && m.phase() == Phase.BASE) return m.value();
        }
        return stat.defaultValue;
    }

    public boolean canUpgrade(Stat stat) {
        return upgradeableStats.contains(stat);
    }
}

package com.example.grimoiremod.glyphs;

import java.util.List;

import static com.example.grimoiremod.glyphs.Stat.*;
import static com.example.grimoiremod.glyphs.StatMod.*;

/**
 * === ТАБЛИЦА БАЛАНСА ===
 * Меняй числа здесь — больше нигде.
 *

public final class GrimoireEffects {
    private GrimoireEffects() {}

    public static void apply(SpellParser.Element element, StatBlock formStats,
                             LivingEntity target, float critFactor, Entity caster, Entity directEntity) {
        if (element == null || formStats == null || target == null) return;
        String id = element.instance().typeId();
        Glyph g = Glyphs.byId(id);
        if (g == null) return;

        // СТАВИМ NBT МЕТКУ (чтобы отследить убийство даже после горения)
        if (caster != null) {
            target.getPersistentData().putString("GrimoireLastElement", id);
            target.getPersistentData().putUUID("GrimoireLastOwner", caster.getUUID());
        }

        int biggerCount = 0;
        for (String m : element.mods()) if ("bigger".equals(m)) biggerCount++;

        double baseDamage = SpellAggregator.elementBaseDamage(element, formStats);
        double baseHeal   = SpellAggregator.elementBaseHeal(element, formStats);
        int durationTicks = (int)Math.round(g.baseStat(Stat.DURATION) * 20.0);

        float finalDamage = (float)(baseDamage * critFactor);
        float finalHeal   = (float)(baseHeal   * critFactor);

        DamageSource ds = caster != null 
                ? target.damageSources().indirectMagic(caster, directEntity != null ? directEntity : caster)
                : target.damageSources().magic();

        switch (id) {
            case "fire", "fireball" -> {
                if (finalDamage > 0) target.hurt(ds, finalDamage);
                target.setRemainingFireTicks(Math.max(target.getRemainingFireTicks(), durationTicks));
            }
            case "ice" -> {
                if (finalDamage > 0) target.hurt(ds, finalDamage);
                target.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, durationTicks, 0));
            }
            case "water" -> {
                if (finalDamage > 0) target.hurt(ds, finalDamage);
                target.setRemainingFireTicks(0);
                target.addEffect(new MobEffectInstance(GrimoireMod.WET, durationTicks, 0));
                if (target.level() instanceof ServerLevel sl) {
                    sl.sendParticles(ParticleTypes.SPLASH,
                            target.getX(), target.getY() + target.getBbHeight() * 0.5, target.getZ(),
                            12, 0.3, 0.3, 0.3, 0.05);
                }
            }
            case "heal" -> {
                if (finalHeal > 0) target.heal(finalHeal);
                if (target.level() instanceof ServerLevel sl) {
                    sl.sendParticles(ParticleTypes.HEART,
                            target.getX(), target.getY() + target.getBbHeight() * 0.5, target.getZ(),
                            4, 0.3, 0.3, 0.3, 0.05);
                }
            }
            case "speed" -> target.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SPEED, durationTicks, Math.max(0, biggerCount)));
            case "slow" -> target.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, durationTicks, Math.max(0, biggerCount)));
        }
    }

----- src/main/java/com/example/grimoiremod/glyphs/SpellAggregator.java -----
package com.example.grimoiremod.glyphs;

import com.example.grimoiremod.data.GlyphInstance;
import com.example.grimoiremod.data.GlyphPage;
import com.example.grimoiremod.data.SpellParser;

import java.util.EnumMap;
import java.util.List;

/**
 * Единственный источник правды по урону/лечению.
 * Формовые статы (через StatBlock) и поэлементные (через elementBase*).
 */
public final class SpellAggregator {
    private static final int MAX_SPLIT = 2;
    private SpellAggregator() {}

    public static StatBlock aggregate(GlyphPage page) {
        if (page == null || page.isToc()) return StatBlock.defaults();

        SpellParser.Parsed parsed = SpellParser.parse(page);
        if (parsed.formId() == null) return StatBlock.defaults();
        Glyph form = Glyphs.byId(parsed.formId());
        if (form == null) return StatBlock.defaults();

        int splitCount = 0, focusCount = 0, spreadCount = 0;
        for (String modId : parsed.formMods()) {
            switch (modId) {
                case "split"  -> splitCount++;
                case "focus"  -> focusCount++;
                case "spread" -> spreadCount++;
            }
        }
        splitCount = Math.min(splitCount, MAX_SPLIT);
        if (focusCount > 0 && spreadCount > 0) { focusCount = 0; spreadCount = 0; }

        EnumMap<Stat, Double> stats = new EnumMap<>(Stat.class);
        for (Stat s : Stat.values()) stats.put(s, s.defaultValue);

        applyBase(stats, form.stats());
        applyModifier(stats, "split",  splitCount);
        applyModifier(stats, "focus",  focusCount);
        applyModifier(stats, "spread", spreadCount);

        // Применяем вложенные очки характеристик от глифа формы
        GlyphInstance formInst = null;
        for (GlyphInstance inst : page.slots) {
            if (inst != null && inst.typeId().equals(parsed.formId())) {
                formInst = inst;
                break;
            }
        }
        
        if (formInst != null) {
            // Перезарядка: каждое очко уменьшает время на 0.05 сек
            int cdPoints = formInst.getPoints(Stat.COOLDOWN);
            if (cdPoints > 0) {
                double baseCd = stats.get(Stat.COOLDOWN);
                stats.put(Stat.COOLDOWN, Math.max(0.1, baseCd - cdPoints * 0.05));
            }
            // Сюда в будущем можно добавить другие статы формы, если нужно
        }

        return new StatBlock(stats);
    }

    private static void applyBase(EnumMap<Stat, Double> stats, List<StatMod> mods) {
        for (StatMod m : mods) {
            if (m.phase() == Phase.BASE) stats.put(m.stat(), m.value());
        }
    }

    private static void applyModifier(EnumMap<Stat, Double> stats, String glyphId, int count) {
        if (count <= 0) return;
        Glyph g = Glyphs.byId(glyphId);
        if (g == null) return;
        for (int i = 0; i < count; i++) {
            for (StatMod m : g.stats()) {
                if (m.phase() == Phase.ADD) {
                    stats.merge(m.stat(), m.value(), Double::sum);
                } else if (m.phase() == Phase.MULT) {
                    stats.merge(m.stat(), m.value(), (a, b) -> a * b);
                }
            }
        }
    }

    /** Сумма ADD DAMAGE от модификаторов элемента (generic, не только bigger). */
    public static double elementModDamageBonus(SpellParser.Element e) {
        double bonus = 0;
        for (String m : e.mods()) {
            Glyph mg = Glyphs.byId(m);
            if (mg == null) continue;
            for (StatMod sm : mg.stats()) {
                if (sm.stat() == Stat.DAMAGE && sm.phase() == Phase.ADD) bonus += sm.value();
            }
        }
        return bonus;
    }

    /** Итоговый урон элемента ДО крита: base + моды + поинты + DAMAGE_BONUS формы. */
    public static double elementBaseDamage(SpellParser.Element e, StatBlock formStats) {
        Glyph g = Glyphs.byId(e.instance().typeId());
        if (g == null) return 0;
        return g.baseStat(Stat.DAMAGE)
             + elementModDamageBonus(e)
             + e.instance().getPoints(Stat.DAMAGE)
             + formStats.get(Stat.DAMAGE_BONUS);
    }

    /** Итоговое лечение элемента ДО крита. */
    public static double elementBaseHeal(SpellParser.Element e, StatBlock formStats) {
        Glyph g = Glyphs.byId(e.instance().typeId());
        if (g == null) return 0;
        return g.baseStat(Stat.HEAL)
             + e.instance().getPoints(Stat.HEAL)
             + formStats.get(Stat.DAMAGE_BONUS);
    }
}

package com.example.grimoiremod.glyphs;

import net.minecraft.network.chat.Component;

public enum Stat {
    FALLBACK_DAMAGE(Kind.NUMBER,  2.0),
    CRIT_CHANCE    (Kind.PERCENT, 0.0),
    CRIT_MULT      (Kind.PERCENT, 0.0),
    COOLDOWN       (Kind.SECONDS, 1.0),
    COUNT          (Kind.INT,     1.0),
    ANGLE          (Kind.MULT,    1.0),
    LIFE           (Kind.MULT,    1.0),
    DAMAGE_BONUS   (Kind.NUMBER,  0.0),
    DAMAGE         (Kind.NUMBER,  0.0),
    HEAL           (Kind.NUMBER,  0.0),
    DURATION       (Kind.SECONDS, 0.0);

    public enum Kind { NUMBER, INT, PERCENT, SECONDS, MULT }

    public final Kind kind;
    public final double defaultValue;

    Stat(Kind kind, double defaultValue) {
        this.kind = kind;
        this.defaultValue = defaultValue;
    }

    public Component getDisplayName() {
        return Component.translatable("stat.grimoiremod." + name().toLowerCase());
    }

    public String format(double value, Phase phase) {
        String prefix = "";
        String body;
        switch (kind) {
            case NUMBER -> {
                if (phase == Phase.ADD && value >= 0) prefix = "+";
                body = trimNumber(value);
            }
            case INT -> {
                if (phase == Phase.ADD && value >= 0) prefix = "+";
                body = String.valueOf((int)Math.round(value));
            }
            case PERCENT -> {
                int pct = (int)Math.round(value * 100);
                if (phase == Phase.ADD && pct >= 0) prefix = "+";
                body = pct + "%";
            }
            case SECONDS -> {
                if (phase == Phase.ADD && value >= 0) prefix = "+";
                body = trimNumber(value) + "s";
            }
            case MULT -> {
                prefix = "x";
                body = trimNumber(value);
            }
            default -> body = String.valueOf(value);
        }
        return prefix + body;
    }

    // для целых — без хвоста, для дробных — до 2 знаков
    public static String trimNumber(double v) {
        if (v == Math.rint(v) && Math.abs(v) < 1e15) return String.valueOf((long)v);
        String s = String.format("%.2f", v);
        s = s.replaceAll("\\.?0+$", "");
        return s;
    }
}

package com.example.grimoiremod.item;

import com.example.grimoiremod.data.GlyphPage;
import com.example.grimoiremod.data.GrimoireData;
import com.example.grimoiremod.data.SkillTreeData;
import com.example.grimoiremod.data.SpellParser;
import com.example.grimoiremod.entity.SpellProjectile;
import com.example.grimoiremod.glyphs.GrimoireEffects;
import com.example.grimoiremod.glyphs.SpellAggregator;
import com.example.grimoiremod.glyphs.Stat;
import com.example.grimoiremod.glyphs.StatBlock;
import net.minecraft.world.InteractionHand;
import net.minecraft.world.InteractionResultHolder;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.item.Item;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.Level;
import net.minecraft.world.phys.AABB;
import net.minecraft.world.phys.Vec3;

import java.util.List;

public class GrimoireItem extends Item {
    private static final float BASE_SPREAD_PER_SPLIT = 0.15f;

    public GrimoireItem(Properties props) { super(props); }

    @Override
    public InteractionResultHolder<ItemStack> use(Level level, Player player, InteractionHand hand) {
        ItemStack stack = player.getItemInHand(hand);
        if (player.isSpectator()) return InteractionResultHolder.pass(stack);
        if (!level.isClientSide) {
            float cd = cast(level, player, stack);
            if (cd > 0f) player.getCooldowns().addCooldown(this, Math.max(1, (int)(cd * 20)));
        }
        return InteractionResultHolder.sidedSuccess(stack, level.isClientSide);
    }

    private float cast(Level level, Player player, ItemStack stack) {
        int active = GrimoireData.getActivePage(stack);
        GlyphPage page = GrimoireData.getPage(stack, active);

        SpellParser.Parsed parsed = SpellParser.parse(page);
        if (parsed.formId() == null) return 0f;

        StatBlock sb = SpellAggregator.aggregate(page);
        List<SpellParser.Element> elements = parsed.elements();

        switch (parsed.formId()) {
            case "projectile", "fireball" -> castProjectile(level, player, sb, elements);
            case "self"       -> castSelf(player, sb, elements);
            case "touch"      -> castTouch(level, player, sb, elements);
        }
        
        // Перезарядка теперь полностью высчитывается внутри SpellAggregator
        return sb.getFloat(Stat.COOLDOWN);
    }

    private void castProjectile(Level level, Player player, StatBlock sb, List<SpellParser.Element> elements) {
        int count = sb.getInt(Stat.COUNT);
        float angle = sb.getFloat(Stat.ANGLE);
        float spreadPerSplit = BASE_SPREAD_PER_SPLIT * angle;
        float critFactor = rollCrit(player, sb);
        Vec3 look = player.getLookAngle();
        for (int i = 0; i < count; i++) {
            double yawOffset = (i - (count - 1) / 2.0) * spreadPerSplit;
            Vec3 dir = look.yRot((float) yawOffset);
            level.addFreshEntity(new SpellProjectile(level, player, elements, sb, critFactor, dir));
        }
    }

    private void castSelf(Player player, StatBlock sb, List<SpellParser.Element> elements) {
        float critFactor = rollCrit(player, sb);
        for (SpellParser.Element e : elements) GrimoireEffects.apply(e, sb, player, critFactor, player, null);
    }

    private void castTouch(Level level, Player player, StatBlock sb, List<SpellParser.Element> elements) {
        AABB box = player.getBoundingBox().inflate(3);
        List<LivingEntity> targets = level.getEntitiesOfClass(LivingEntity.class, box, e -> e != player);
        if (targets.isEmpty()) return;
        LivingEntity closest = null;
        double best = Double.MAX_VALUE;
        for (LivingEntity t : targets) {
            double d = t.distanceToSqr(player);
            if (d < best) { best = d; closest = t; }
        }
        if (closest == null) return;
        float critFactor = rollCrit(player, sb);
        for (SpellParser.Element e : elements) GrimoireEffects.apply(e, sb, closest, critFactor, player, null);
    }

    private static float rollCrit(Player player, StatBlock sb) {
        boolean crit = player.getRandom().nextFloat() < sb.getFloat(Stat.CRIT_CHANCE);
        return crit ? (1f + sb.getFloat(Stat.CRIT_MULT)) : 1f;
    }
}

import net.neoforged.neoforge.event.entity.living.LivingDropsEvent;
import net.neoforged.neoforge.event.entity.player.PlayerEvent;

import java.util.UUID;

@EventBusSubscriber(modid = GrimoireMod.MOD_ID)
public class ModEvents {

    private static final String SAVED_GRIMOIRE = "SavedGrimoire";

    @SubscribeEvent
    public static void addToCreativeTab(BuildCreativeModeTabContentsEvent event) {
        if (event.getTabKey() == CreativeModeTabs.TOOLS_AND_UTILITIES) {
            event.accept(GrimoireMod.GRIMOIRE);
            event.accept(GrimoireMod.CREATIVE_GRIMOIRE);
        }
    }

    @SubscribeEvent
    public static void onItemToss(ItemTossEvent event) {
        if (!(event.getEntity().getItem().getItem() instanceof GrimoireItem)) return;
        if (event.getEntity().getTarget() == null) {
            if (event.getPlayer().getInventory().add(event.getEntity().getItem())) {
                event.setCanceled(true);
            }
        }
    }

    @SubscribeEvent
    public static void onLivingDeath(LivingDeathEvent event) {
        LivingEntity deadEntity = event.getEntity();
        if (deadEntity.level().isClientSide()) return;

        ServerPlayer player = null;
        String element = null;

        // 1. Проверяем прямой удар снарядом
        Entity direct = event.getSource().getDirectEntity();
        if (direct instanceof SpellProjectile proj) {
            if (proj.getOwner() instanceof ServerPlayer p) {
                player = p;
                element = proj.primaryElement();
            }
        } 
        // 2. Если умер от горения/яда — читаем NBT метку
        else {
            CompoundTag data = deadEntity.getPersistentData();
            if (data.contains("GrimoireLastOwner")) {
                UUID ownerUUID = data.getUUID("GrimoireLastOwner");
                player = deadEntity.getServer().getPlayerList().getPlayer(ownerUUID);
                element = data.getString("GrimoireLastElement");
            }
        }

        if (player == null || element == null || element.isEmpty()) return;

        String elementId = element.equals("fireball") ? "fire" : element;
        String nodeId = "grimoiremod:element_" + elementId + "_t1";

        int slot = GrimoireLocator.findFirst(player);
        if (slot < 0) {
            player.sendSystemMessage(Component.literal("§c[DEBUG] Гримуар не найден в инвентаре!"));
            return;
        }
        
        ItemStack stack = player.getInventory().getItem(slot);
        if (!(stack.getItem() instanceof GrimoireItem)) return;

        if (!SkillTreeData.isUnlocked(stack, nodeId)) {
            player.sendSystemMessage(Component.literal("§c[DEBUG] Узел " + elementId + " не изучен!"));
            return;
        }

        float freeMastery = 0.1f;
        float nodeExp = 0.9f;

        SkillTreeData.addNodeExp(stack, nodeId, nodeExp, 100);
        SkillTreeData.addPoints(stack, freeMastery);
        
        player.getInventory().setChanged();
        player.containerMenu.broadcastChanges();

        player.sendSystemMessage(Component.literal("§a+0.1 mastery (" + elementId + ")"), true);
    }

    @SubscribeEvent
    public static void onLivingDrops(LivingDropsEvent event) {
        if (!(event.getEntity() instanceof ServerPlayer player)) return;
        event.getDrops().removeIf(drop -> {
            if (drop.getItem().getItem() instanceof GrimoireItem) {
                Tag tag = drop.getItem().save(player.level().registryAccess());
                CompoundTag wrapper = new CompoundTag();
                wrapper.put("Item", tag);
                player.getPersistentData().put(SAVED_GRIMOIRE, wrapper);
                return true;
            }
            return false;
        });
    }

    @SubscribeEvent
    public static void onPlayerClone(PlayerEvent.Clone event) {
        if (!event.isWasDeath()) return;
        CompoundTag data = event.getOriginal().getPersistentData();
        if (data.contains(SAVED_GRIMOIRE)) {
            CompoundTag wrapper = data.getCompound(SAVED_GRIMOIRE);
            Tag itemTag = wrapper.get("Item");
            if (itemTag != null) {
                ItemStack saved = ItemStack.CODEC.parse(
                        event.getEntity().level().registryAccess().createSerializationContext(NbtOps.INSTANCE),
                        itemTag
                ).result().orElse(ItemStack.EMPTY);
                if (!saved.isEmpty()) {
                    event.getEntity().getInventory().add(saved);
                }
            }
            data.remove(SAVED_GRIMOIRE);
        }
    }
}

package com.example.grimoiremod.network;

import com.example.grimoiremod.data.GrimoireData;
import com.example.grimoiremod.glyphs.Stat;
import com.example.grimoiremod.item.GrimoireItem;
import com.example.grimoiremod.menu.GlyphWorkbenchMenu;
import net.minecraft.network.FriendlyByteBuf;
import net.minecraft.network.codec.StreamCodec;
import net.minecraft.network.protocol.common.custom.CustomPacketPayload;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.world.item.ItemStack;
import net.neoforged.neoforge.network.handling.IPayloadContext;

public record AllocatePointPacket(int itemSlot, int pageIdx, int slotIdx, Stat stat, int delta, boolean isBank)
        implements CustomPacketPayload {

    public static final Type<AllocatePointPacket> TYPE =
            new Type<>(ResourceLocation.fromNamespaceAndPath("grimoiremod", "allocate_point"));

    public static final StreamCodec<FriendlyByteBuf, AllocatePointPacket> CODEC = StreamCodec.of(
        (buf, pkt) -> {
            buf.writeVarInt(pkt.itemSlot());
            buf.writeVarInt(pkt.pageIdx());
            buf.writeVarInt(pkt.slotIdx());
            buf.writeUtf(pkt.stat().name());
            buf.writeVarInt(pkt.delta());
            buf.writeBoolean(pkt.isBank());
        },
        buf -> {
            int itemSlot = buf.readVarInt();
            int pageIdx = buf.readVarInt();
            int slotIdx = buf.readVarInt();
            String statName = buf.readUtf(64);
            int delta = buf.readVarInt();
            boolean isBank = buf.readBoolean();

            Stat stat;
            int safeDelta = Math.max(-20, Math.min(20, delta));
            try { stat = Stat.valueOf(statName); }
            catch (IllegalArgumentException e) { stat = Stat.DAMAGE; safeDelta = 0; }
            return new AllocatePointPacket(itemSlot, pageIdx, slotIdx, stat, safeDelta, isBank);
        }
    );

    @Override public Type<? extends CustomPacketPayload> type() { return TYPE; }

    public static void handle(AllocatePointPacket pkt, IPayloadContext ctx) {
        ctx.enqueueWork(() -> {
            ItemStack stack;
            if (ctx.player().containerMenu instanceof GlyphWorkbenchMenu menu) {
                stack = menu.getGrimoire();
            } else {
                int slot = pkt.itemSlot();
                if (slot < 0 || slot >= ctx.player().getInventory().getContainerSize()) return;
                stack = ctx.player().getInventory().getItem(slot);
            }

            if (stack.getItem() instanceof GrimoireItem) {
                if (pkt.isBank()) {
                    GrimoireData.allocatePointToBank(stack, pkt.slotIdx(), pkt.stat(), pkt.delta());
                } else {
                    GrimoireData.allocatePoint(stack, pkt.pageIdx(), pkt.slotIdx(), pkt.stat(), pkt.delta());
                }
                ctx.player().containerMenu.broadcastChanges();
            }
        });
    }
}

public record GrimoireActionPacket(int itemSlot, String action, int[] ints, String[] strings)
        implements CustomPacketPayload {

    private static final int MAX_INTS = 32;
    private static final int MAX_STRINGS = 16;
    private static final int MAX_STR_LEN = 256;

    public static final Type<GrimoireActionPacket> TYPE =
            new Type<>(ResourceLocation.fromNamespaceAndPath("grimoiremod", "grimoire_action"));

    public static final StreamCodec<FriendlyByteBuf, GrimoireActionPacket> CODEC = StreamCodec.of(
        (buf, pkt) -> {
            buf.writeVarInt(pkt.itemSlot());
            buf.writeUtf(pkt.action());
            buf.writeVarInt(pkt.ints().length);
            for (int i : pkt.ints()) buf.writeVarInt(i);
            buf.writeVarInt(pkt.strings().length);
            for (String s : pkt.strings()) buf.writeUtf(s);
        },
        buf -> {
            int itemSlot = buf.readVarInt();
            String action = buf.readUtf(32);
            int intCount = buf.readVarInt();
            if (intCount < 0 || intCount > MAX_INTS) throw new DecoderException("Invalid int count");
            int[] ints = new int[intCount];
            for (int i = 0; i < intCount; i++) ints[i] = buf.readVarInt();
            int strCount = buf.readVarInt();
            if (strCount < 0 || strCount > MAX_STRINGS) throw new DecoderException("Invalid string count");
            String[] strings = new String[strCount];
            for (int i = 0; i < strCount; i++) strings[i] = buf.readUtf(MAX_STR_LEN);
            return new GrimoireActionPacket(itemSlot, action, ints, strings);
        }
    );

    @Override public Type<? extends CustomPacketPayload> type() { return TYPE; }

    }

        StringBuilder sb = new StringBuilder();
        for (var e : allocs.entrySet()) {
            if (sb.length() > 0) sb.append(",");
            sb.append(e.getKey().name()).append(":").append(e.getValue());
        }
        return new GrimoireActionPacket(slot, "placeGlyph", new int[]{page, row}, new String[]{typeId, sb.toString()});
    }

    }
