public class GlyphWorkbenchBlock extends Block {
    public GlyphWorkbenchBlock(Properties props) {
        super(props);
    }

    public static BlockBehaviour.Properties defaultProps() {
        return BlockBehaviour.Properties.of()
                .strength(2.5F)
                .sound(SoundType.WOOD);
    }

    @Override
    public InteractionResult useWithoutItem(BlockState state, Level level, BlockPos pos, Player player, BlockHitResult hit) {
        if (!level.isClientSide) {
            int slot = GrimoireLocator.findFirst(player);
            if (slot < 0) {
                player.displayClientMessage(Component.translatable("gui.grimoiremod.need_grimoire"), true);
                return InteractionResult.CONSUME;
            }

            // Фикс: если открыто оглавление, переключаем на первое заклинание
            ItemStack stack = player.getInventory().getItem(slot);
            if (stack.getItem() instanceof GrimoireItem) {
                int active = GrimoireData.getActivePage(stack);
                GlyphPage page = GrimoireData.getPage(stack, active);
                if (page == null || page.isToc()) {
                    GrimoireData.switchPage(stack, true);
                }
            }

            MenuProvider provider = new SimpleMenuProvider(
                    (id, inv, p) -> new GlyphWorkbenchMenu(id, inv, slot),
                    Component.translatable("gui.grimoiremod.workbench")
            );
            ((ServerPlayer) player).openMenu(provider, buf -> buf.writeVarInt(slot));
        }
        return InteractionResult.sidedSuccess(level.isClientSide);
    }
}

@EventBusSubscriber(modid = GrimoireMod.MOD_ID, value = Dist.CLIENT)
public class ClientEvents {

    @SubscribeEvent
    public static void onClientTick(ClientTickEvent.Post event) {
        Minecraft mc = Minecraft.getInstance();
        Player player = mc.player;
        if (player == null) return;

        if (mc.screen instanceof GrimoireScreen gs && gs.isNameFieldFocused()) {
            while (KeyBindings.PREV_SPELL.consumeClick()) { }
            while (KeyBindings.NEXT_SPELL.consumeClick()) { }
            while (KeyBindings.SKILL_TREE.consumeClick()) { }
            return;
        }

        while (KeyBindings.PREV_SPELL.consumeClick()) switchPage(player, false);
        while (KeyBindings.NEXT_SPELL.consumeClick()) switchPage(player, true);
        while (KeyBindings.SKILL_TREE.consumeClick()) openSkillTree(player);
    }

    private static void switchPage(Player player, boolean next) {
        Minecraft mc = Minecraft.getInstance();
        int slot;
        if (mc.screen instanceof GrimoireScreen gs) {
            slot = gs.getSlotId();
        } else {
            slot = GrimoireLocator.findFirst(player);
            if (slot < 0) return;
        }
        ItemStack stack = player.getInventory().getItem(slot);
        if (!(stack.getItem() instanceof GrimoireItem)) return;
        PacketDistributor.sendToServer(GrimoireActionPacket.switchPage(slot, next));
    }

    private static void openSkillTree(Player player) {
        Minecraft mc = Minecraft.getInstance();
        int slot;
        if (mc.screen instanceof GrimoireScreen gs) slot = gs.getSlotId();
        else slot = GrimoireLocator.findFirst(player);
        
        // Теперь открывается всегда, даже если гримуара нет (slot = -1)
        mc.setScreen(new SkillTreeScreen(slot));
    }

    @SubscribeEvent
    public static void onScreenInit(ScreenEvent.Init.Post event) {
        Minecraft mc = Minecraft.getInstance();
        if (mc.player == null) return;
        int slot = GrimoireLocator.findFirst(mc.player);
        if (slot < 0) return;

        if (event.getScreen() instanceof InventoryScreen inv) {
            int guiLeft = (inv.width - 176) / 2;
            int guiTop = (inv.height - 166) / 2;
            int x = Math.max(4, guiLeft - 64);
            int y = guiTop + 4;
            event.addListener(Button.builder(
                    Component.translatable("gui.grimoiremod.grimoire"),
                    b -> mc.setScreen(new GrimoireScreen(slot))
            ).pos(x, y).size(60, 20).build());
            return;
        }

        if (event.getScreen() instanceof CreativeModeInventoryScreen) {
            event.addListener(Button.builder(
                    Component.translatable("gui.grimoiremod.grimoire"),
                    b -> mc.setScreen(new GrimoireScreen(slot))
            ).pos(4, 4).size(60, 20).build());
        }
    }

    public static int findGrimoireSlot(Player player) {
        return GrimoireLocator.findFirst(player);
    }

import java.util.HashMap;
import java.util.Map;

public final class ClientGlyphIcons {
    private static final Map<String, Boolean> CACHE = new HashMap<>();

    private ClientGlyphIcons() {}

    public static void clearCache() { CACHE.clear(); }

    public static boolean has(String glyphId) {
        if (glyphId == null || glyphId.isEmpty()) return false;
        return CACHE.computeIfAbsent(glyphId, k -> {
            try {
                var mc = Minecraft.getInstance();
                if (mc.getResourceManager() == null) return false;
                return mc.getResourceManager().getResource(path(k)).isPresent();
            } catch (Exception e) {
                return false;
            }
        });
    }

    public static ResourceLocation path(String glyphId) {
        return ResourceLocation.fromNamespaceAndPath(GrimoireMod.MOD_ID,
                "textures/gui/glyph/" + glyphId + ".png");
    }

    public static void draw(GuiGraphics gfx, Font font, Glyph g, int x, int y, int size) {
        int cx = x + (size - 16) / 2;
        int cy = y + (size - 16) / 2;
        if (has(g.id())) {
            gfx.blit(path(g.id()), cx, cy, 0, 0, 16, 16, 16, 16);
        } else {
            gfx.fill(cx, cy, cx + 16, cy + 16, g.color());
            String initial = g.getDisplayName().getString().substring(0, 1).toUpperCase();
            gfx.drawString(font, initial, cx + 8 - font.width(initial) / 2, cy + 4, 0xFFFFFFFF, false);
        }
    }

    @EventBusSubscriber(modid = GrimoireMod.MOD_ID, value = Dist.CLIENT)
    public static class ReloadListener {
        @SubscribeEvent
        public static void onRegisterReloadListeners(RegisterClientReloadListenersEvent event) {
            event.registerReloadListener(
                new SimplePreparableReloadListener<Void>() {
                    @Override
                    protected Void prepare(ResourceManager resourceManager, ProfilerFiller profiler) {
                        return null;
                    }
                    @Override
                    protected void apply(Void object, ResourceManager resourceManager, ProfilerFiller profiler) {
                        clearCache();
                    }
                }
            );
        }
    }
}

package com.example.grimoiremod.client;

import com.example.grimoiremod.data.GlyphInstance;
import com.example.grimoiremod.glyphs.Glyph;
import com.example.grimoiremod.glyphs.Glyphs;
import com.example.grimoiremod.glyphs.Phase;
import com.example.grimoiremod.glyphs.Stat;
import com.example.grimoiremod.glyphs.StatMod;
import net.minecraft.ChatFormatting;
import net.minecraft.network.chat.Component;

import java.util.ArrayList;
import java.util.List;

public final class GlyphTooltip {
    private GlyphTooltip() {}

    public static List<Component> build(GlyphInstance inst, String errKey) {
        List<Component> tooltip = new ArrayList<>();
        if (inst == null) return tooltip;
        Glyph g = Glyphs.byId(inst.typeId());
        if (g == null) return tooltip;

        tooltip.add(Component.literal(". ").withStyle(s -> s.withColor(g.color() & 0xFFFFFF))
                .append(g.getDisplayName().copy().withStyle(ChatFormatting.WHITE)));
        if (inst.totalPoints() > 0) {
            tooltip.add(Component.translatable("gui.grimoiremod.points_spent", inst.totalPoints())
                    .withStyle(ChatFormatting.GOLD));
        }
        
        for (StatMod sm : g.stats()) {
            if (sm.phase() == Phase.BASE) {
                double val = sm.value();
                int pts = inst.getPoints(sm.stat());
                
                // Применяем вложенные очки
                if (sm.stat() == Stat.COOLDOWN) {
                    val = Math.max(0.1, val - pts * 0.05);
                } else {
                    val += pts;
                }
                
                tooltip.add(Component.literal("  ").append(
                        Component.empty()
                                .append(sm.stat().getDisplayName())
                                .append(": " + sm.stat().format(val, sm.phase()))
                ).withStyle(ChatFormatting.GRAY));
            } else {
                tooltip.add(Component.literal("  ").append(sm.tooltip()).withStyle(ChatFormatting.GRAY));
            }
        }
        
        if (g.stats().isEmpty()) {
            tooltip.add(Component.translatable("gui.grimoiremod.no_stats").withStyle(ChatFormatting.DARK_GRAY));
        }
        if (errKey != null) {
            tooltip.add(Component.literal("  ").append(Component.translatable(errKey))
                    .withStyle(ChatFormatting.RED));
        }
        return tooltip;
    }
}

            if (selectedSlot >= 0 && selectedSlot < page.slotCount && page.slots[selectedSlot] != null) {
                targetInst = page.slots[selectedSlot];
            } else if (selectedBankInst != null) {
                targetInst = selectedBankInst;
                isBankTarget = true;
            }

            if (targetInst != null) {
                Glyph g = Glyphs.byId(targetInst.typeId());
                if (g != null) {
                    int textX = leftPos + 190, textY = topPos + 20;
                    String prefix = isBankTarget ? Component.translatable("gui.grimoiremod.bank_prefix").getString() : "";
                    gfx.drawString(font, prefix + g.getDisplayName().getString(), textX, textY, 0xFF3A2E20, false);
                    gfx.drawString(font, Component.translatable("gui.grimoiremod.points", targetInst.totalPoints()).getString(),

                    gfx.fill(textX, textY + 24, leftPos + 330, textY + 25, 0xFF8D6E63);

                    List<Stat> stats = new ArrayList<>();
                    for (Stat cand : Stat.values()) {
                        if (g.canUpgrade(cand)) stats.add(cand);
                    }
                    
                    for (int s = 0; s < stats.size(); s++) {
                        Stat stat = stats.get(s);
                        int pts = targetInst.getPoints(stat);
                        int y = textY + 35 + s * 24;
                        gfx.drawString(font, stat.getDisplayName().getString() + ": " + pts, textX, y + 2, 0xFF333333, false);

                        int plusX = leftPos + 280, minusX = leftPos + 305;
                        boolean hoverPlus = inRect(mouseX, mouseY, plusX, y, BUTTON_W, BUTTON_H);
                        boolean hoverMinus = inRect(mouseX, mouseY, minusX, y, BUTTON_W, BUTTON_H);
                        boolean isAdvanced = g.advanced();
                        boolean canAdd = !isAdvanced && (GrimoireData.getSpentPoints(book, stat) < GrimoireData.getStatBudget(book, stat));
                        boolean canSub = !isAdvanced && (pts > 0);

                        int plusFill = canAdd ? (hoverPlus ? 0xFF77AA77 : 0xFF555555) : 0xFF333333;
                        gfx.fill(plusX, y, plusX + BUTTON_W, y + BUTTON_H, plusFill);
                        gfx.drawString(font, "+", plusX + 4, y + 2, canAdd ? 0xFFFFFFFF : 0xFF888888, false);

    @Override
    public boolean mouseClicked(double mx, double my, int button) {
        ItemStack book = menu.getGrimoire();
        if (!(book.getItem() instanceof GrimoireItem)) return super.mouseClicked(mx, my, button);

        int active = GrimoireData.getActivePage(book);
        GlyphPage page = GrimoireData.getPage(book, active);
        if (page == null || page.isToc()) return super.mouseClicked(mx, my, button);

        int navX = leftPos + 60, navY = topPos + 22;
        if (inRect(mx, my, navX, navY, 12, 12)) {
            PacketDistributor.sendToServer(GrimoireActionPacket.switchPage(grimoireSlot, false));
            selectedSlot = -1; selectedBankInst = null; return true;
        }
        if (inRect(mx, my, navX + 40, navY, 12, 12)) {
            PacketDistributor.sendToServer(GrimoireActionPacket.switchPage(grimoireSlot, true));
            selectedSlot = -1; selectedBankInst = null; return true;
        }

        int startX = leftPos + 18, startY = topPos + 60;
        for (int i = 0; i < WORKBENCH_SLOTS; i++) {
            if (i >= page.slotCount) continue;
            int x = startX + (i % 5) * (SLOT_SIZE + SLOT_GAP);
            int y = startY + (i / 5) * (SLOT_SIZE + SLOT_GAP);
            if (inRect(mx, my, x, y, SLOT_SIZE, SLOT_SIZE)) {
                if (button == 1) {
                    if (page.slots[i] != null) {
                        PacketDistributor.sendToServer(GrimoireActionPacket.removeGlyph(grimoireSlot, active, i));
                        selectedSlot = -1;
                    }
                } else {
                    if (page.slots[i] != null) {
                        selectedSlot = (selectedSlot == i) ? -1 : i;
                        selectedBankInst = null;
                    } else if (selectedBankInst != null) {
                        PacketDistributor.sendToServer(GrimoireActionPacket.placeGlyph(
                                grimoireSlot, active, i, selectedBankInst.typeId(), selectedBankInst.allocations()));
                        selectedBankInst = null;
                    }
                }
                return true;
            }
        }

        BankView bankView = buildBankView(book);
        List<GlyphInstance> groupList = bankView.groups();
        int bankX = leftPos + 20, bankY = topPos + 150;
        int maxScroll = Math.max(0, groupList.size() - (BANK_COLS * BANK_ROWS_VISIBLE));

        if (groupList.size() > BANK_COLS * BANK_ROWS_VISIBLE) {
            int scrollX = leftPos + 162;
            if (inRect(mx, my, scrollX, bankY, 10, 10)) { bankScroll = Math.max(0, bankScroll - BANK_COLS); return true; }
            if (inRect(mx, my, scrollX, bankY + 40, 10, 10)) { bankScroll = Math.min(maxScroll, bankScroll + BANK_COLS); return true; }
        }

        for (int gi = bankScroll; gi < groupList.size() && gi < bankScroll + BANK_COLS * BANK_ROWS_VISIBLE; gi++) {
            int localIdx = gi - bankScroll;
            int col = localIdx % BANK_COLS, row = localIdx / BANK_COLS;
            int x = bankX + col * (CELL_SIZE + GAP);
            int y = bankY + row * (CELL_SIZE + GAP);
            if (inRect(mx, my, x, y, CELL_SIZE, CELL_SIZE)) {
                GlyphInstance inst = groupList.get(gi);
                selectedBankInst = (selectedBankInst != null && selectedBankInst.equals(inst)) ? null : inst;
                selectedSlot = -1;
                return true;
            }
        }

        List<GlyphInstance> rawBank = GrimoireData.getBankList(book);
        GlyphInstance targetInst = null;
        boolean isBankTarget = false;
        if (selectedSlot >= 0 && selectedSlot < page.slotCount && page.slots[selectedSlot] != null) {
            targetInst = page.slots[selectedSlot];
        } else if (selectedBankInst != null) {
            targetInst = selectedBankInst;
            isBankTarget = true;
        }

        if (targetInst != null) {
            List<Stat> stats = new ArrayList<>();
            Glyph g = Glyphs.byId(targetInst.typeId());
            for (Stat cand : Stat.values()) {
                if (g.canUpgrade(cand)) stats.add(cand);
            }
            
            int textY = topPos + 20;
            boolean isAdvanced = g != null && g.advanced();
            for (int s = 0; s < stats.size(); s++) {
                int y = textY + 35 + s * 24;
                int plusX = leftPos + 280, minusX = leftPos + 305;
                
                int targetIdx = -1;
                if (isBankTarget) {
                    targetIdx = rawBank.indexOf(selectedBankInst);
                } else {
                    targetIdx = selectedSlot;
                }

                if (inRect(mx, my, plusX, y, BUTTON_W, BUTTON_H)) {
                    if (!isAdvanced && GrimoireData.canAllocate(book, stats.get(s), 1) && targetIdx != -1) {
                        PacketDistributor.sendToServer(new AllocatePointPacket(
                                grimoireSlot, active, targetIdx, stats.get(s), 1, isBankTarget));
                    }
                    return true;
                }
                if (inRect(mx, my, minusX, y, BUTTON_W, BUTTON_H)) {
                    if (!isAdvanced && targetIdx != -1) {
                        PacketDistributor.sendToServer(new AllocatePointPacket(
                                grimoireSlot, active, targetIdx, stats.get(s), -1, isBankTarget));
                    }
                    return true;
                }
            }
            int resetY = textY + 165;
            if (inRect(mx, my, leftPos + 190, resetY, 60, 16)) {
                if (isBankTarget) {
                    int idx = rawBank.indexOf(selectedBankInst);
                    if (idx != -1) PacketDistributor.sendToServer(GrimoireActionPacket.resetGlyphInBank(grimoireSlot, idx));
                } else {
                    PacketDistributor.sendToServer(GrimoireActionPacket.resetGlyph(grimoireSlot, active, selectedSlot));
                }
                return true;
            }
        }
        return super.mouseClicked(mx, my, button);
    }

    private boolean inRect(double mx, double my, int x, int y, int w, int h) {
        return mx >= x && mx < x + w && my >= y && my < y + h;
    }

    private ItemStack stack() {
        var p = Minecraft.getInstance().player;
        if (p == null) return ItemStack.EMPTY;
        ItemStack s = p.getInventory().getItem(itemSlotId);
        return s.getItem() instanceof GrimoireItem ? s : ItemStack.EMPTY;
    }

    private Snapshot snapshot() {
        ItemStack s = stack();
        if (s.isEmpty()) return null;
        CustomData cd = s.get(DataComponents.CUSTOM_DATA);
        CompoundTag tag = cd != null ? cd.copyTag() : new CompoundTag();
        if (tag.equals(cachedTag) && cachedSnap != null) return cachedSnap;

        int pageCount = GrimoireData.getPageCount(s);
        int active    = GrimoireData.getActivePage(s);
        List<GlyphPage> pages = new ArrayList<>(pageCount);
        int[] spellNums = new int[pageCount];
        int n = 0;
        boolean hasToc = false;
        for (int i = 0; i < pageCount; i++) {
            GlyphPage p = GrimoireData.getPage(s, i);
            pages.add(p);
            if (p != null && p.isSpell()) n++;
            if (p != null && p.isToc()) hasToc = true;
            spellNums[i] = n;
        }
        Map<GlyphInstance, Integer> bank = null;
        GlyphPage activePage = (active >= 0 && active < pages.size()) ? pages.get(active) : null;
        if (activePage != null && activePage.isSpell()) {
            bank = GrimoireData.getBankGrouped(s);
        }

        SpellParser.Parsed parsed = null;
        List<Component> described = null;
        if (activePage != null && activePage.isSpell()) {
            parsed = SpellParser.parse(activePage);
            described = SpellDescriber.describeLines(activePage);
        }

        cachedTag = tag;
        cachedSnap = new Snapshot(pageCount, active, pages, spellNums, bank, hasToc, parsed, described);
        return cachedSnap;
    }

    private void renderBookmarks(GuiGraphics gfx, int mx, int my, Snapshot snap, int active) {
        int bx = bookLeft + BOOK_W + COVER_PAD + 2;
        int by0 = bookTop + 8;

        int maxBookmarks = 10;
        int bh = (BOOK_H - 16) / maxBookmarks;
        int favCount = 0;
        hoverBookmark = -1;

        for (int i = 0; i < snap.pageCount() && favCount < maxBookmarks; i++) {
            GlyphPage p = snap.page(i);
            if (p == null || !p.favorite) continue;

            int by = by0 + favCount * bh;
            boolean hover = inRect(mx, my, bx, by, 22, bh - 1);
            if (hover) hoverBookmark = i;

            int color = (i == active) ? brighten(p.bookmarkColor) : p.bookmarkColor;
            int border = hover ? 0xFFFFFFFF : 0xFF2E1C10;
            gfx.fill(bx - 1, by - 1, bx + 23, by + bh, border);
            gfx.fill(bx, by, bx + 22, by + bh - 1, color);

            String label = p.isToc() ? "T" : String.valueOf(snap.spellNumber(i));
            int textColor = getContrastColor(color);
            gfx.drawString(font, label, bx + 11 - font.width(label) / 2, by + bh / 2 - 4, textColor, false);

            if (hover) {
                String name = p.isToc()
                        ? Component.translatable("gui.grimoiremod.toc").getString()
                        : (p.name == null || p.name.isBlank()
                            ? Component.translatable("gui.grimoiremod.spell", snap.spellNumber(i)).getString()
                            : p.name);
                int tw = font.width(name);
                gfx.fill(bx + 22, by, bx + 22 + tw + 8, by + bh - 1, color);
                gfx.fill(bx + 22 + tw + 8, by, bx + 22 + tw + 9, by + bh - 1, border);
                gfx.drawString(font, name, bx + 26, by + bh / 2 - 4, textColor, false);
            }

            favCount++;
        }
    }

    private int getContrastColor(int color) {
        int r = (color >> 16) & 0xFF;
        int g = (color >> 8) & 0xFF;
        int b = color & 0xFF;
        int yiq = ((r * 299) + (g * 587) + (b * 114)) / 1000;
        return (yiq >= 128) ? 0xFF000000 : 0xFFFFFFFF;
    }

    private void renderTocPage(GuiGraphics gfx, int mx, int my, Snapshot snap) {
        gfx.drawString(font, Component.translatable("gui.grimoiremod.toc").getString(),
                leftPageX + 8, leftPageY + 4, INK, false);
        gfx.fill(leftPageX + 8, leftPageY + 18, leftPageX + PAGE_W - 8, leftPageY + 19, INK_DIM);

        int count = snap.pageCount();
        int rowY = leftPageY + TOC_TOP;
        boolean hasIndicator = tocDragFrom >= 0 && tocDropTarget >= 0 && tocDragMoved;
        int indicatorY = -1;

        if (hasIndicator) {
            int to = computeToPage(tocDragFrom, tocDropTarget, tocDropAfter);
            if (to != tocDragFrom) {
                int scanY = leftPageY + TOC_TOP;
                for (int i = 0; i < count; i++) {
                    GlyphPage p = snap.page(i);
                    if (p == null || p.isToc()) continue;
                    if (i == tocDropTarget) {
                        indicatorY = tocDropAfter ? scanY + TOC_ROW_H - 2 : scanY;
                        break;
                    }
                    scanY += TOC_ROW_H;
                }
            }
        }

        for (int i = 0; i < count; i++) {
            GlyphPage p = snap.page(i);
            if (p == null) continue;
            if (p.isToc()) continue;

            boolean isDragging = (i == tocDragFrom);
            boolean hover = inRect(mx, my, leftPageX + 8, rowY, PAGE_W - 16, TOC_ROW_H - 2);

            int bg;
            if (isDragging && tocDragMoved) bg = PAGE_BG_DRAG;
            else if (hover) bg = PAGE_BG_HOVER;
            else bg = PAGE_BG_ALT;

            gfx.fill(leftPageX + 8, rowY, leftPageX + PAGE_W - 8, rowY + TOC_ROW_H - 2, bg);
            gfx.fill(leftPageX + 12, rowY + 4, leftPageX + 16, rowY + TOC_ROW_H - 6, p.bookmarkColor);

            int num = snap.spellNumber(i);
            String label = Component.translatable("gui.grimoiremod.spell", num).getString();
            if (p.name != null && !p.name.isBlank()) label += ": " + p.name;
            int textColor = (isDragging && tocDragMoved) ? INK_DIM : INK;
            gfx.drawString(font, label, leftPageX + 22, rowY + 5, textColor, false);

            String favStr = p.favorite ? "*" : "-";
            gfx.drawString(font, favStr, leftPageX + PAGE_W - 20, rowY + 5,
                    p.favorite ? 0xFFFFAA00 : INK_DIM, false);

            rowY += TOC_ROW_H;
        }

        if (indicatorY >= 0) {
            gfx.fill(leftPageX + 8, indicatorY - 1, leftPageX + PAGE_W - 8, indicatorY + 1, TOC_GOLD);
        }

        String hint = tocDragFrom >= 0 && tocDragMoved
                ? Component.translatable("gui.grimoiremod.drop_hint").getString()
                : Component.translatable("gui.grimoiremod.drag_hint").getString();
        gfx.drawString(font, hint, leftPageX + 8, leftPageY + PAGE_H - 20, INK_DIM, false);
    }
